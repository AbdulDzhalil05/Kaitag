package com.kaitag.vpn

import android.content.Context
import android.content.Intent
import android.net.ConnectivityManager
import android.net.NetworkCapabilities
import android.util.Log
import androidx.test.ext.junit.runners.AndroidJUnit4
import androidx.test.platform.app.InstrumentationRegistry
import hev.htproxy.TProxyService
import org.junit.After
import org.junit.Assert.assertFalse
import org.junit.Assert.assertNull
import org.junit.Assert.assertTrue
import org.junit.Before
import org.junit.Test
import org.junit.runner.RunWith

/**
 * Regression coverage for the start policy of [KaitagVpnService].
 *
 * After the process is killed the system recreates the sticky service with a null
 * intent, and in general any start request may arrive without ACTION_CONNECT. Such a
 * start must never bring Xray / hev / TUN up — only ACTION_CONNECT may do that.
 *
 * intent == null cannot be delivered from outside the framework, so the negative test
 * drives the very same onStartCommand branch with a start request that has no action.
 * Waiting is done through the service's own ACTION_TUNNEL_STATE reports (see
 * [VpnXrayHevChainTest.StateListener]) instead of sleeps: the first report is the
 * barrier "the request was processed", and its value is the verdict.
 */
@RunWith(AndroidJUnit4::class)
class KaitagVpnServiceStartPolicyTest {

    private val context: Context
        get() = InstrumentationRegistry.getInstrumentation().targetContext

    private lateinit var state: VpnXrayHevChainTest.StateListener

    @Before
    fun setUp() {
        VpnXrayHevChainTest.stopEverything()
        state = VpnXrayHevChainTest.StateListener(context)
        state.register()
    }

    @After
    fun tearDown() {
        state.unregister()
        VpnXrayHevChainTest.stopEverything()
    }

    /** 1+2. Без ACTION_CONNECT сервис не должен поднимать Xray / hev / VPN. */
    @Test
    fun startWithoutConnectActionDoesNotStartTunnel() {
        context.startForegroundService(Intent(context, KaitagVpnService::class.java))

        assertHandledWithoutTunnel("start with no action")

        // Тот же branch покрывает и пересоздание после убийства процесса
        // (intent == null): никакого автозапуска без явной команды.
        val unknown = VpnXrayHevChainTest.StateListener(context)
        unknown.register()
        try {
            context.startForegroundService(
                Intent(context, KaitagVpnService::class.java)
                    .setAction("com.kaitag.vpn.NOT_A_CONNECT_COMMAND"),
            )
            assertHandledWithoutTunnel("start with an unknown action", unknown)
        } finally {
            unknown.unregister()
        }
    }

    /** 3. ACTION_CONNECT действительно поднимает туннель. */
    @Test
    fun explicitConnectActionStartsTunnel() {
        ensureConsentAndIdleTunnel()

        context.sendBroadcast(
            VpnXrayHevChainTest.testControlIntent(VpnTestControlReceiver.ACTION_START),
        )
        assertTrue(
            "ACTION_CONNECT must start the tunnel; lastError=${state.lastError}",
            state.awaitRunning(true),
        )
        assertTrue("Xray must be running after ACTION_CONNECT", XrayRuntime.isRunning())
        assertTrue("hev must be running after ACTION_CONNECT", TProxyService.TProxyIsRunning())
        assertTrue("xray SOCKS port must be reported", state.xrayPort > 0)
        assertTrue("VPN network must appear after ACTION_CONNECT", awaitVpnNetwork(true))
        Log.i(TAG, "ACTION_CONNECT brought the chain up on port ${state.xrayPort}")
    }

    /** 4. ACTION_DISCONNECT останавливает туннель и не оставляет ничего работающего. */
    @Test
    fun explicitDisconnectActionStopsTunnel() {
        ensureConsentAndIdleTunnel()
        context.sendBroadcast(
            VpnXrayHevChainTest.testControlIntent(VpnTestControlReceiver.ACTION_START),
        )
        assertTrue(
            "setup: ACTION_CONNECT must start the tunnel; lastError=${state.lastError}",
            state.awaitRunning(true),
        )

        context.sendBroadcast(
            VpnXrayHevChainTest.testControlIntent(VpnTestControlReceiver.ACTION_STOP),
        )
        assertTrue(
            "ACTION_DISCONNECT must stop the tunnel",
            state.awaitRunning(false),
        )
        assertTunnelDown("after ACTION_DISCONNECT")
    }

    /**
     * The service answers the request it was given; the answer must be running=false
     * with nothing started. The first report is the barrier, no fixed sleeps involved.
     */
    private fun assertHandledWithoutTunnel(
        what: String,
        listener: VpnXrayHevChainTest.StateListener = state,
    ) {
        val first = listener.awaitFirstState()
        assertTrue(
            "$what must be answered by the service with running=false, got=$first " +
                "(null = no answer, true = tunnel started without ACTION_CONNECT)",
            first == false,
        )
        assertTunnelDown(what)
        assertFalse(
            "$what must never be reported as running=true",
            listener.awaitRunning(true, 2_000),
        )
    }

    private fun assertTunnelDown(what: String) {
        assertFalse("Xray must be down $what", XrayRuntime.isRunning())
        assertFalse("hev must be down $what", TProxyService.TProxyIsRunning())
        // VPN Network снимается системой асинхронно после закрытия tun-fd сервисом,
        // поэтому ждём исчезновения (как chain-тест) вместо мгновенного снимка allNetworks.
        assertTrue("VPN network must be absent $what", awaitVpnNetwork(false, timeoutMs = 10_000))
    }

    /**
     * VPN consent is granted through the system dialog, and accepting it makes
     * MainActivity start the tunnel by itself. Wait for that start and stop it, so the
     * command under test is what brings the chain up.
     */
    private fun ensureConsentAndIdleTunnel() {
        if (!VpnXrayHevChainTest.vpnConsentGranted()) {
            VpnXrayHevChainTest.grantVpnConsentThroughSystemDialog()
            assertTrue(
                "VPN consent missing after the system dialog",
                VpnXrayHevChainTest.vpnConsentGranted(),
            )
            VpnXrayHevChainTest.awaitServiceStarted()
        }
        VpnXrayHevChainTest.stopEverything()
        assertTunnelDown("before the command under test")
    }

    private fun vpnNetwork(): NetworkCapabilities? {
        val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        for (network in cm.allNetworks) {
            val caps = cm.getNetworkCapabilities(network) ?: continue
            if (caps.hasTransport(NetworkCapabilities.TRANSPORT_VPN)) return caps
        }
        return null
    }

    private fun awaitVpnNetwork(expected: Boolean, timeoutMs: Long = 30_000): Boolean {
        val deadline = System.currentTimeMillis() + timeoutMs
        while (System.currentTimeMillis() < deadline) {
            if ((vpnNetwork() != null) == expected) return true
            Thread.sleep(500)
        }
        return false
    }

    companion object {
        private const val TAG = "StartPolicyTest"
    }
}

package com.kaitag.vpn

import android.util.Log
import libXray.LibXray
import org.junit.After
import org.junit.Assert.assertEquals
import org.junit.Assert.assertTrue
import org.junit.Before
import org.junit.Test
import org.junit.runner.RunWith
import androidx.test.ext.junit.runners.AndroidJUnit4
import java.io.ByteArrayOutputStream
import java.io.InputStream
import java.io.OutputStream
import java.net.InetAddress
import java.net.InetSocketAddress
import java.net.ServerSocket
import java.net.Socket
import org.json.JSONObject

@RunWith(AndroidJUnit4::class)
class LibXraySocks5Test {

    private var socksPort = 0

    @Before
    fun setUp() {
        Log.i(TAG, "LibXray xrayVersion -> ${invoke("xrayVersion")}")
        assertEquals("Xray must not be running before the test", false, isRunning())
        socksPort = pickFreePort()
        Log.i(TAG, "chosen local SOCKS5 port: $socksPort")
    }

    @After
    fun tearDown() {
        val stopResponse = invoke("stopXray")
        Log.i(TAG, "LibXray stopXray -> $stopResponse")
        Log.i(TAG, "LibXray getXrayState (after stop) -> ${invoke("getXrayState")}")
        LibXray.resetDNS()
    }

    private fun invoke(method: String, payload: String = "{}"): String {
        val request = """{"apiVersion":3,"method":"$method","payload":$payload}"""
        val response = LibXray.invoke(request)
        Log.i(TAG, "LibXray $method -> $response")
        return response
    }

    private fun dataOf(response: String): JSONObject {
        val parsed = JSONObject(response)
        assertTrue("LibXray call failed: $response", parsed.getBoolean("success"))
        assertEquals("LibXray error: $response", "", parsed.getString("error"))
        val data = parsed.get("data")
        return if (data == JSONObject.NULL) JSONObject() else data as JSONObject
    }

    private fun isRunning(): Boolean {
        val response = LibXray.invoke(
            """{"apiVersion":3,"method":"getXrayState","payload":{}}"""
        )
        val parsed = JSONObject(response)
        assertTrue("getXrayState failed: $response", parsed.getBoolean("success"))
        return parsed.getJSONObject("data").getBoolean("running")
    }

    /**
     * Uses the official getFreePorts method as a candidate source, then confirms the
     * candidate can actually be bound on loopback before handing it to Xray.
     */
    private fun pickFreePort(): Int {
        val response = invoke("getFreePorts", """{"count":3,"excludePorts":[$FALLBACK_PORT]}""")
        val candidates = dataOf(response).getJSONArray("ports")
        Log.i(TAG, "getFreePorts candidates: $candidates")
        for (i in 0 until candidates.length()) {
            val port = candidates.getInt(i)
            if (isBindable(port)) {
                Log.i(TAG, "getFreePorts candidate $port verified bindable")
                return port
            }
            Log.i(TAG, "getFreePorts candidate $port already in use, skipping")
        }
        require(isBindable(FALLBACK_PORT)) { "fallback port $FALLBACK_PORT is not free" }
        Log.i(TAG, "falling back to verified-free port $FALLBACK_PORT")
        return FALLBACK_PORT
    }

    private fun isBindable(port: Int): Boolean =
        try {
            ServerSocket(port, 1, InetAddress.getByName("127.0.0.1")).use { true }
        } catch (e: Exception) {
            false
        }

    private fun buildConfig(port: Int): String {
        val config = JSONObject()
            .put("log", JSONObject().put("loglevel", "warning"))
            .put(
                "inbounds",
                org.json.JSONArray().put(
                    JSONObject()
                        .put("tag", "socks-in")
                        .put("protocol", "socks")
                        .put("listen", "127.0.0.1")
                        .put("port", port)
                        .put(
                            "settings",
                            JSONObject().put("auth", "noauth").put("udp", true)
                        )
                        .put("sniffing", JSONObject().put("enabled", false))
                )
            )
            .put(
                "outbounds",
                org.json.JSONArray().put(
                    JSONObject()
                        .put("tag", "direct")
                        .put("protocol", "freedom")
                        .put("settings", JSONObject())
                )
            )
        return config.toString()
    }

    @Test
    fun socks5FreedomProxyServesHttpTraffic() {
        val config = buildConfig(socksPort)
        Log.i(TAG, "xray config: $config")

        val payload = JSONObject().put("xrayJson", config).toString()
        dataOf(invoke("testXray", payload))
        val runResponse = invoke("runXray", payload)
        dataOf(runResponse)

        val stateAfterStart = invoke("getXrayState")
        assertTrue(
            "Xray should be running: $stateAfterStart",
            JSONObject(stateAfterStart).getJSONObject("data").getBoolean("running")
        )

        // Control: confirm the emulator itself has working outbound internet access.
        val directStatus = runCatching { directHttpStatus(CONTROL_HOST) }
            .onSuccess { Log.i(TAG, "control direct HTTP $CONTROL_HOST -> $it") }
            .onFailure { Log.w(TAG, "control direct HTTP $CONTROL_HOST failed: ${it.message}") }

        // Each SOCKS5 CONNECT creates its own tunnel, so use a fresh socket per target.
        var socksConnectOk = false
        var httpOk = false
        val failures = mutableListOf<String>()
        for (target in HTTP_TARGETS) {
            val host = target.first
            val remotePort = target.second
            try {
                val socket = awaitSocksConnect(remotePort = remotePort, host = host)
                Log.i(TAG, "SOCKS5 CONNECT established via 127.0.0.1:$socksPort -> $host:$remotePort")
                socksConnectOk = true
                val statusLine = httpGetOverSocket(socket, host, remotePort)
                Log.i(TAG, "HTTP via SOCKS5 $host:$remotePort -> $statusLine")
                assertTrue("Expected an HTTP status line, got: $statusLine", statusLine.startsWith("HTTP/1."))
                httpOk = true
                socket.close()
                break
            } catch (e: Exception) {
                failures += "$host:$remotePort -> ${e.message}"
                Log.w(TAG, "HTTP via SOCKS5 $host:$remotePort failed: ${e.message}")
            }
        }
        assertTrue("SOCKS5 CONNECT never succeeded: $failures", socksConnectOk)
        assertTrue("No HTTP target succeeded through SOCKS5: $failures (control direct: $directStatus)", httpOk)

        val stopResponse = invoke("stopXray")
        dataOf(stopResponse)

        val stateAfterStop = invoke("getXrayState")
        assertTrue(
            "Xray must be stopped: $stateAfterStop",
            !JSONObject(stateAfterStop).getJSONObject("data").getBoolean("running")
        )
        }

    private fun awaitSocksConnect(
        remotePort: Int,
        host: String,
        timeoutMs: Long = 15_000
    ): Socket {
        val deadline = System.currentTimeMillis() + timeoutMs
        var lastError: Exception? = null
        while (System.currentTimeMillis() < deadline) {
            try {
                return socksHandshake(socksPort, host, remotePort, timeoutMs.toInt())
            } catch (e: Exception) {
                lastError = e
                Thread.sleep(250)
            }
        }
        throw IllegalStateException(
            "SOCKS5 listener on $socksPort never completed CONNECT to $host:$remotePort",
            lastError
        )
    }

    private fun directHttpStatus(host: String): String {
        Socket().use { socket ->
            socket.connect(InetSocketAddress(host, 80), 15_000)
            socket.soTimeout = 15_000
            return httpGetOverSocket(socket, host, 80)
        }
    }

    /**
     * Minimal RFC 1928 SOCKS5 client: greeting, no-auth, CONNECT, then reply parsing.
     */
    private fun socksHandshake(
        port: Int,
        host: String,
        remotePort: Int,
        timeoutMs: Int
    ): Socket {
        val socket = Socket()
        socket.connect(InetSocketAddress(InetAddress.getByName("127.0.0.1"), port), timeoutMs)
        socket.soTimeout = timeoutMs
        val out = socket.getOutputStream()
        val input = socket.getInputStream()

        out.write(byteArrayOf(0x05, 0x01, 0x00))
        out.flush()
        val greeting = ByteArray(2)
        readFully(input, greeting, "greeting")
        check(greeting[0].toInt() and 0xFF == 0x05) { "bad SOCKS version 0x${greeting[0].toInt() and 0xFF}" }
        check(greeting[1].toInt() and 0xFF == 0x00) {
            "SOCKS5 server rejected no-auth, method=0x${greeting[1].toInt() and 0xFF}"
        }

        val hostBytes = host.toByteArray(Charsets.US_ASCII)
        val request = ByteArrayOutputStream()
        request.write(0x05)
        request.write(0x01)
        request.write(0x00)
        request.write(0x03)
        request.write(hostBytes.size)
        request.write(hostBytes)
        request.write((remotePort ushr 8) and 0xFF)
        request.write(remotePort and 0xFF)
        out.write(request.toByteArray())
        out.flush()

        val head = ByteArray(4)
        readFully(input, head, "reply head")
        val reply = head[1].toInt() and 0xFF
        val atyp = head[3].toInt() and 0xFF
        val addressLength = when (atyp) {
            0x01 -> 4
            0x04 -> 16
            0x03 -> {
                val len = ByteArray(1)
                readFully(input, len, "domain length")
                len[0].toInt() and 0xFF
            }
            else -> throw IllegalStateException("unknown SOCKS ATYP $atyp")
        }
        readFully(input, ByteArray(addressLength), "bound address")
        readFully(input, ByteArray(2), "bound port")
        check(reply == 0x00) { "SOCKS5 CONNECT to $host failed, REP=$reply (${replyName(reply)})" }
        return socket
    }

    private fun httpGetOverSocket(socket: Socket, host: String, port: Int): String {
        val request = "GET / HTTP/1.1\r\nHost: $host\r\nUser-Agent: KaitagVPN-LibXray-Test\r\nAccept: */*\r\nConnection: close\r\n\r\n"
        val out: OutputStream = socket.getOutputStream()
        out.write(request.toByteArray(Charsets.US_ASCII))
        out.flush()

        val input = socket.getInputStream()
        val statusLine = StringBuilder()
        var byte = input.read()
        if (byte == -1) throw IllegalStateException("empty response from $host")
        while (byte != -1 && byte != '\n'.code) {
            if (byte != '\r'.code) statusLine.append(byte.toChar())
            byte = input.read()
        }
        return statusLine.toString().trim()
    }

    private fun readFully(input: InputStream, buffer: ByteArray, what: String) {
        var offset = 0
        while (offset < buffer.size) {
            val read = input.read(buffer, offset, buffer.size - offset)
            if (read == -1) throw IllegalStateException("unexpected EOF while reading $what")
            offset += read
        }
    }

    private fun replyName(reply: Int): String = when (reply) {
        0x00 -> "succeeded"
        0x01 -> "general SOCKS server failure"
        0x02 -> "connection not allowed by ruleset"
        0x03 -> "network unreachable"
        0x04 -> "host unreachable"
        0x05 -> "connection refused"
        0x06 -> "TTL expired"
        0x07 -> "command not supported"
        0x08 -> "address type not supported"
        else -> "unknown"
    }

        private const val CONTROL_HOST = "cp.cloudflare.com"
        private val HTTP_TARGETS = listOf(

    private fun awaitSocksConnect(
        remotePort: Int,
        host: String,
        timeoutMs: Long = 20_000
    ): Socket {
        val deadline = System.currentTimeMillis() + timeoutMs
        var lastError: Exception? = null
        while (System.currentTimeMillis() < deadline) {
            try {
                return socksHandshake(socksPort, host, remotePort, timeoutMs.toInt())
            } catch (e: Exception) {
                lastError = e
                Thread.sleep(250)

        }
        throw IllegalStateException(
            "SOCKS5 listener on $socksPort never completed CONNECT to $host:$remotePort",
            lastError,
        )

    private fun socksHandshake(
        port: Int,
        host: String,
        remotePort: Int,
        timeoutMs: Int
    ): Socket {
        val socket = Socket()
        socket.connect(InetSocketAddress(InetAddress.getByName("127.0.0.1"), port), timeoutMs)
        socket.soTimeout = timeoutMs
        val out = socket.getOutputStream()
        val input = socket.getInputStream()

        out.write(byteArrayOf(0x05, 0x01, 0x00))
        out.flush()
        val greeting = ByteArray(2)
        readFully(input, greeting, "greeting")
        check(greeting[0].toInt() and 0xFF == 0x05) { "bad SOCKS version" }
        check(greeting[1].toInt() and 0xFF == 0x00) { "SOCKS5 server rejected no-auth" }

        val hostBytes = host.toByteArray(Charsets.US_ASCII)
        val request = ByteArrayOutputStream()
        request.write(0x05)
        request.write(0x01)
        request.write(0x00)
        request.write(0x03)
        request.write(hostBytes.size)
        request.write(hostBytes)
        request.write((remotePort ushr 8) and 0xFF)
        request.write(remotePort and 0xFF)
        out.write(request.toByteArray())
        out.flush()

        val head = ByteArray(4)
        readFully(input, head, "reply head")
        val reply = head[1].toInt() and 0xFF
        val atyp = head[3].toInt() and 0xFF
        val addressLength = when (atyp) {
            0x01 -> 4
            0x04 -> 16
            0x03 -> {
                val len = ByteArray(1)
                readFully(input, len, "domain length")
                len[0].toInt() and 0xFF
            }
            else -> throw IllegalStateException("unknown SOCKS ATYP $atyp")
        }
        readFully(input, ByteArray(addressLength), "bound address")
        readFully(input, ByteArray(2), "bound port")
        check(reply == 0x00) { "SOCKS5 CONNECT to $host failed, REP=$reply" }
        return socket
    }

    private fun httpGetOverSocket(socket: Socket, host: String, port: Int): String {
        val out: OutputStream = socket.getOutputStream()
        out.write(
            ("GET / HTTP/1.1\r\nHost: $host\r\nUser-Agent: KaitagVPN-VlessTest\r\n" +
                "Accept: */*\r\nConnection: close\r\n\r\n").toByteArray(Charsets.US_ASCII)
        )
        out.flush()
        return readStatusLine(socket.getInputStream())
    }

    private fun readStatusLine(input: InputStream): String {
        val sb = StringBuilder()
        var byte = input.read()
        if (byte == -1) throw IllegalStateException("empty response")
        while (byte != -1 && byte != '\n'.code) {
            if (byte != '\r'.code) sb.append(byte.toChar())
            byte = input.read()
        }
        return sb.toString().trim()
    }

    /** Cloudflare stamps CF-Ray with the serving colo, so it differs when egress changes. */
    private fun readHeaders(input: InputStream): String {
        val text = StringBuilder()
        var byte = input.read()
        if (byte == -1) throw IllegalStateException("empty response")
        while (byte != -1 && byte != '\n'.code) {
            if (byte != '\r'.code) text.append(byte.toChar())
            byte = input.read()
            if (text.contains("\n\n") || text.length > 2048) break
        }
        return text.toString()
    }

    private fun cfRayOf(headers: String): String? =
        Regex("CF-Ray:\\s*(\\S+)", RegexOption.IGNORE_CASE).find(headers)?.groupValues?.get(1)

    private fun readFully(input: InputStream, buffer: ByteArray, what: String) {
        var offset = 0
        while (offset < buffer.size) {
            val read = input.read(buffer, offset, buffer.size - offset)
            if (read == -1) throw IllegalStateException("unexpected EOF while reading $what")
            offset += read
        }
    }

    /** Минимальный DNS A-запрос: заголовок (RD) + qname + тип A + класс IN. */
    private fun buildDnsQuery(transactionId: Int, name: String): ByteArray {
        val qname = ByteArrayOutputStream()
        for (label in name.split('.')) {
            qname.write(label.length)
            qname.write(label.toByteArray(Charsets.US_ASCII))
        }
        qname.write(0)
        val labels = qname.toByteArray()
        val query = ByteArray(12 + labels.size + 4)
        query[0] = (transactionId ushr 8).toByte()
        query[1] = transactionId.toByte()
        query[2] = 0x01 // RD
        query[5] = 0x01 // QDCOUNT = 1
        labels.copyInto(query, 12)
        val qtype = 12 + labels.size
        query[qtype] = 0x00; query[qtype + 1] = 0x01 // A
        query[qtype + 2] = 0x00; query[qtype + 3] = 0x01 // IN
        return query
    }

        return ByteArray(hex.length / 2) { i ->

        }
    }

        val before = trafficBytes()

        // executeShellCommand runs the command directly instead of through a shell, so a pipe
        // cannot be passed inline. Writing the pipeline to a script file and running
        // "sh <file>" sidesteps that: the fetch then runs as uid 2000 (shell), which the VPN
        // routes into the tunnel.
        val downloadIp = InetAddress.getByName(DOWNLOAD_HOST).hostAddress
        val script = java.io.File(context.getExternalFilesDir(null), "kaitag-tun-download.sh")
        // A single nc transfer can be cut short mid-stream, so retry and keep the best count.
        // The threshold stays high on purpose: the delta must be well above background noise.
        script.writeText(
            """
            best=0
            i=0
            while [ ${'$'}i -lt 5 ]; do
              n=${'$'}(printf 'GET $DOWNLOAD_PATH HTTP/1.1\r\nHost: $DOWNLOAD_HOST\r\nConnection: close\r\n\r\n' | nc -w 30 $downloadIp 80 | wc -c)
              echo "attempt ${'$'}i bytes=${'$'}n" >&2
              [ "${'$'}n" -gt "${'$'}best" ] && best=${'$'}n
              [ "${'$'}best" -gt 500000 ] && break
              i=${'$'}((${'$'}i + 1))
            done
            echo ${'$'}best
            """.trimIndent(),
        )
        Log.i(TAG, "6. download script ${script.absolutePath} -> http://$downloadIp$DOWNLOAD_PATH")

        val downloaded = shellUidCommand("sh ${script.absolutePath}", timeoutMs = 60_000)
            ?.trim()
            ?.lines()
            ?.lastOrNull()
            ?.trim()
            ?.toLongOrNull()
        Log.i(TAG, "6. shell downloaded bytes = $downloaded")
        val received = requireNotNull(downloaded) { "the shell download produced no byte count" }
        assertTrue(
            "the shell only received $received bytes of the 1MB file, so it likely failed",
            received > 500_000L,
        )

        val after = awaitTunTraffic(before, timeoutMs = 90_000)
        val delta = after - before
        Log.i(TAG, "6. counters before=$before after=$after delta=$delta")
        assertTrue(
            "hev counters grew by only $delta after a ~1MB download by an external uid " +
                "(before=$before after=$after)",
            delta > 500_000L,
        )

        // --- stop via the test control channel ---
        context.sendBroadcast(testControlIntent(VpnTestControlReceiver.ACTION_STOP))
        val down = state.awaitRunning(false)
        assertTrue("service never reported tunnel running=false after STOP", down)

        assertTrue("TProxyService still running", !TProxyService.TProxyIsRunning())
        assertTrue("Xray still running", !XrayRuntime.isRunning())
        Log.i(TAG, "stopped: TProxyIsRunning=false, Xray running=false")

        assertTrue("VPN network still present after STOP", awaitVpnNetwork(false))
        Log.i(TAG, "stopped: no VPN network left in allNetworks")

        // --- normal network works again ---
        val direct = httpGetDirect(CONTROL_HOST, 80)
        Log.i(TAG, "post-stop direct HTTP: $direct")
        assertTrue("Network broken after VPN stop", direct.startsWith("HTTP/1."))
    }

    /**
     * Listens for the service's ACTION_TUNNEL_STATE / ACTION_TUNNEL_ERROR reports.
     * The app UID cannot run `ip`, so state is read from the service itself.
     */
    internal class StateListener(private val context: Context) {
        @Volatile var xrayPort: Int = 0
            private set

        @Volatile var lastError: String? = null
            private set

        private val seen = mutableSetOf<Boolean>()

        /** Первое состояние после [register]; null — пока ничего не приходило. */
        private var firstState: Boolean? = null

        /** Последнее подтверждённое сервисом состояние (а не «когда-либо виденное»). */
        @Volatile
        private var latest: Boolean? = null

        private val receiver = object : android.content.BroadcastReceiver() {
            override fun onReceive(c: Context, intent: Intent) {
                when (intent.action) {
                    KaitagVpnService.ACTION_TUNNEL_STATE -> {
                        val running = intent.getBooleanExtra(
                            KaitagVpnService.EXTRA_TUNNEL_RUNNING, false
                        )
                        xrayPort = intent.getIntExtra(KaitagVpnService.EXTRA_XRAY_SOCKS_PORT, 0)
                        synchronized(seen) {
                            seen.add(running)
                            if (firstState == null) firstState = running
                        }
                        latest = running
                        Log.i(TAG, "state broadcast: running=$running xrayPort=$xrayPort")
                    }
                    KaitagVpnService.ACTION_TUNNEL_ERROR -> {
                        lastError = intent.getStringExtra(KaitagVpnService.EXTRA_ERROR_MESSAGE)
                        Log.e(TAG, "error broadcast: $lastError")
                    }
                }
            }
        }

        fun register() {
            val filter = android.content.IntentFilter().apply {
                addAction(KaitagVpnService.ACTION_TUNNEL_STATE)
                addAction(KaitagVpnService.ACTION_TUNNEL_ERROR)
            }
            androidx.core.content.ContextCompat.registerReceiver(
                context, receiver, filter, androidx.core.content.ContextCompat.RECEIVER_NOT_EXPORTED
            )
        }

        fun unregister() = runCatching { context.unregisterReceiver(receiver) }

        /**
         * Ждёт, пока последнее подтверждённое сервисом состояние станет [expected].
         * Условие «когда-либо видели» в сравнении недостаточно: broadcast running=false,
         * пришедший ДО старта, не должен удовлетворять «дождаться остановки».
         */
        fun awaitRunning(expected: Boolean, timeoutMs: Long = 30_000): Boolean {
            val deadline = System.currentTimeMillis() + timeoutMs
            while (System.currentTimeMillis() < deadline) {
                if (latest == expected) return true
                if (!expected && lastError != null) return false
                Thread.sleep(500)
            }
            return false
        }

        /**
         * Ждёт самое первое состояние после [register] и возвращает его
         * (null — если сервис ничего не сообщил). В отличие от [awaitRunning] это
         * барьер «запрос обработан», а не «наступило конкретное состояние».
         */
        fun awaitFirstState(timeoutMs: Long = 30_000): Boolean? {
            val deadline = System.currentTimeMillis() + timeoutMs
            while (System.currentTimeMillis() < deadline) {
                synchronized(seen) { firstState?.let { return it } }
                Thread.sleep(200)
            }
            return null
        }
    }

    /**
     * Runs a command with the shell identity through UiAutomation, so the request is
     * issued by uid 2000 instead of the app uid that addDisallowedApplication excludes.
     */
    private fun shellUidCommand(cmd: String, timeoutMs: Long = 30_000): String? = runCatching {
        val automation = InstrumentationRegistry.getInstrumentation().uiAutomation
        val pfd = automation.executeShellCommand(cmd)
        val out = android.os.ParcelFileDescriptor.AutoCloseInputStream(pfd).use { stream ->
            val text = StringBuilder()
            val buffer = ByteArray(8192)
            val deadline = System.currentTimeMillis() + timeoutMs
            while (System.currentTimeMillis() < deadline) {
                // The pipe can hand back 0 without being at EOF, so only -1 ends it.
                val read = runCatching { stream.read(buffer) }.getOrDefault(-1)
                if (read == -1) break
                if (read == 0) {
                    Thread.sleep(50)
                    continue
                }
                text.append(String(buffer, 0, read, Charsets.UTF_8))
            }
            text.toString()
        }
        Log.i(TAG, "shellUid[$cmd] -> ${out.trim().take(400)}")
        out
    }.getOrElse {
        Log.e(TAG, "shellUid[$cmd] threw", it)
        null
    }

    /**
     * The app itself is excluded from the VPN by addDisallowedApplication, so its own
     * default network is never the tunnel. All networks must be scanned instead, and the
     * UID ranges decide whether a given caller is routed into the tunnel.
     */
    private fun vpnNetwork(): NetworkCapabilities? {
        val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        for (network in cm.allNetworks) {
            val caps = cm.getNetworkCapabilities(network) ?: continue
            if (caps.hasTransport(NetworkCapabilities.TRANSPORT_VPN)) return caps
        }
        return null
    }

    /** Records the kernel's view of the tunnel interface and its default route. */
    private fun logTunnelRoutes() {
        Log.i(TAG, "6. tun interface: ${shellUidCommand("ip -o addr show tun0")?.trim()}")
        Log.i(
            TAG,
            "6. default routes: " +
                shellUidCommand("ip route show table all | grep ' 0.0.0.0/0'")?.trim(),
        )
    }

    private fun awaitVpnNetwork(expected: Boolean, timeoutMs: Long = 30_000): Boolean {
        val deadline = System.currentTimeMillis() + timeoutMs
        while (System.currentTimeMillis() < deadline) {
            if ((vpnNetwork() != null) == expected) return true
            Thread.sleep(500)
        }
        return false
    }

    /** hev-socks5-tunnel byte counters; traffic must increase for the chain to be live. */
    private fun trafficBytes(): Long {
        val stats = TProxyService.TProxyGetStats() ?: return -1L
        Log.i(TAG, "TProxyGetStats raw = ${stats.toList()}")
        return stats.sum()
    }

    private fun awaitTunTraffic(baseline: Long, timeoutMs: Long = 45_000): Long {
        val deadline = System.currentTimeMillis() + timeoutMs
        var current = baseline
        while (System.currentTimeMillis() < deadline) {
            current = trafficBytes()
            if (current > baseline) {
                return current
            }
            Thread.sleep(1000)
        }
        return current
    }

    private fun httpGetThroughTun(host: String, port: Int): String =
        httpGet(viaProxy = true, host = host, port = port)

    private fun httpGetDirect(host: String, port: Int): String =
        httpGet(viaProxy = false, host = host, port = port)

    private fun httpGet(viaProxy: Boolean, host: String, port: Int): String {
        val socket = if (viaProxy) {
            socksHandshake(XrayRuntime.socksPort, host, port, 20_000)
        } else {
            Socket().apply { connect(InetSocketAddress(host, port), 20_000) }
        }
        socket.use {
            it.soTimeout = 20_000
            val out = it.getOutputStream()
            out.write(httpGetRequest(host).toByteArray(Charsets.US_ASCII))
            out.flush()
            return readStatusLine(it.getInputStream())
        }
    }

    private fun httpGetRequest(host: String) =
        "GET / HTTP/1.1\r\nHost: $host\r\nUser-Agent: KaitagVPN-Chain-Test\r\n" +
            "Accept: */*\r\nConnection: close\r\n\r\n"

    private fun socksHandshake(port: Int, host: String, remotePort: Int, timeoutMs: Int): Socket {
        val socket = Socket()
        socket.connect(InetSocketAddress(InetAddress.getByName("127.0.0.1"), port), timeoutMs)
        socket.soTimeout = timeoutMs
        val out: OutputStream = socket.getOutputStream()
        val input: InputStream = socket.getInputStream()

        out.write(byteArrayOf(0x05, 0x01, 0x00))
        out.flush()
        val greeting = ByteArray(2)
        readFully(input, greeting, "greeting")
        check(greeting[1].toInt() and 0xFF == 0x00) { "SOCKS5 no-auth rejected" }

        val hostBytes = host.toByteArray(Charsets.US_ASCII)
        val request = ByteArrayOutputStream()
        request.write(0x05); request.write(0x01); request.write(0x00); request.write(0x03)
        request.write(hostBytes.size); request.write(hostBytes)
        request.write((remotePort ushr 8) and 0xFF); request.write(remotePort and 0xFF)
        out.write(request.toByteArray()); out.flush()

        val head = ByteArray(4)
        readFully(input, head, "reply")
        val reply = head[1].toInt() and 0xFF
        val atyp = head[3].toInt() and 0xFF
        val addressLength = when (atyp) {
            0x01 -> 4
            0x04 -> 16
            0x03 -> {
                val len = ByteArray(1); readFully(input, len, "domain len"); len[0].toInt() and 0xFF
            }
            else -> throw IllegalStateException("unknown ATYP $atyp")
        }
        readFully(input, ByteArray(addressLength), "bound addr")
        readFully(input, ByteArray(2), "bound port")
        check(reply == 0x00) { "SOCKS5 CONNECT to $host failed, REP=$reply" }
        return socket
    }

    private fun readStatusLine(input: InputStream): String {
        val sb = StringBuilder()
        var byte = input.read()
        if (byte == -1) throw IllegalStateException("empty response")
        while (byte != -1 && byte != '\n'.code) {
            if (byte != '\r'.code) sb.append(byte.toChar())
            byte = input.read()
        }
        return sb.toString().trim()
    }

    private fun readFully(input: InputStream, buffer: ByteArray, what: String) {
        var offset = 0
        while (offset < buffer.size) {
            val read = input.read(buffer, offset, buffer.size - offset)
            if (read == -1) throw IllegalStateException("unexpected EOF reading $what")
            offset += read
        }
    }

    private fun shell(cmd: String): String? = runCatching {
        val p = ProcessBuilder("sh", "-c", cmd).redirectErrorStream(true).start()
        val out = p.inputStream.bufferedReader().use { it.readText() }
        Log.i(TAG, "shell[$cmd] exit=${p.waitFor()} out=${out.take(400)}")
        out
    }.getOrElse {
        Log.e(TAG, "shell[$cmd] threw", it)
        null
    }

         * Launches the app's existing VPN consent activity and accepts the system dialog.
         * Uses the app UI already present (MainActivity -> prepareAndStartVpn), no shell
         * permission tricks: the user-facing consent flow is the real mechanism.
         */
        internal fun grantVpnConsentThroughSystemDialog() {
            val instrumentation = InstrumentationRegistry.getInstrumentation()
            val device = UiDevice.getInstance(instrumentation)

            // launchMainActivity: startActivitySync deadlocks when MainActivity is already
            // running (the test shares the app's main thread), so use a monitor when needed.
            val monitor = instrumentation.addMonitor(MainActivity::class.java.name, null, false)
            instrumentation.startActivitySync(
                Intent(context, MainActivity::class.java)
                    .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP)
            )
            val activity = monitor.waitForActivityWithTimeout(10_000)
            Log.i(TAG, "consent: MainActivity = $activity")
            val startButton = device.wait(
                Until.findObject(By.res("com.kaitag.vpn:id/startVpnButton")),
                15_000,
            )
            Log.i(TAG, "consent: startVpnButton visible = ${startButton != null}")

            // The system VPN confirmation dialog animates in; retry until consent sticks.
            val deadline = System.currentTimeMillis() + 60_000
            var attempts = 0
            var logged = false
            while (System.currentTimeMillis() < deadline) {
                if (vpnConsentGranted()) {
                    Log.i(TAG, "consent: granted after $attempts dialog click(s)")
                    return
                }

                // The dialog may not be on screen yet; the start button launches it.
                val startButton = device.findObject(By.res("com.kaitag.vpn:id/startVpnButton"))
                if (startButton != null && startButton.isEnabled) {
                    try {
                        Log.i(TAG, "consent: tapping startVpnButton (attempt $attempts)")
                        startButton.click()
                        attempts++
                    } catch (e: androidx.test.uiautomator.StaleObjectException) {
                        Log.i(TAG, "consent: stale start button, retrying")
                    }
                }

                device.waitForIdle(2_000)
                val okButton = try {
                    device.findObject(By.res("android:id/button1"))?.takeIf { it.isEnabled }
                        ?: device.findObject(By.text("OK"))
                        // The app legitimately asks for POST_NOTIFICATIONS after the
                        // first connect (full-screen GrantPermissionsActivity on modern
                        // SDKs); dismiss that dialog too, either way is fine for the VPN.
                        ?: device.findObject(By.text("Allow"))
                        ?: device.findObject(By.text("Разрешить"))
                        ?: device.findObject(By.text("Don't allow"))
                        ?: device.findObject(By.text("Не разрешать"))
                } catch (e: androidx.test.uiautomator.StaleObjectException) {
                    Log.i(TAG, "consent: stale while reading dialog, retrying")
                    null
                }
                if (okButton != null) {
                    if (!logged) {
                        logWindowObjects(device)
                        logged = true
                    }
                    // The dialog re-binds while animating, so nodes go stale; retry on fresh ones.
                    try {
                        Log.i(TAG, "consent: clicking dialog button '${okButton.text}'")
                        okButton.click()
                        attempts++
                    } catch (e: androidx.test.uiautomator.StaleObjectException) {
                        Log.i(TAG, "consent: stale dialog node, retrying")
                    }
                } else {
                    Log.i(TAG, "consent: dialog button not visible yet")
                }
                Thread.sleep(1_000)
            }
            Log.e(TAG, "consent: not granted after $attempts click(s)")
        }
    }

/**
 * Regression coverage for the lifecycle used by the network handover path.
 *
 * The instrumentation environment cannot switch between real Mobile and Wi-Fi, so the
 * handover *decision* logic is covered by JVM unit tests
 * (see app/src/test/.../NetworkHandoverDeciderTest.kt). What can and must be verified on
 * a device is the part the handover depends on: Xray start -> stop -> start and the
 * repeated restart must leave exactly one consistent instance behind, with a working
 * SOCKS5 endpoint.
 */
@RunWith(AndroidJUnit4::class)
class XrayNetworkHandoverRuntimeTest {

    @Before
    fun setUp() {
        drainTunnelQuietly()
        assertTrue("Xray must be stopped before the test", !XrayRuntime.isRunning())
    }

    @After
    fun tearDown() {
        XrayRuntime.stop()
        Log.i(TAG, "tearDown: running=${XrayRuntime.isRunning()} port=${XrayRuntime.socksPort}")
    }

    private fun drainTunnelQuietly() {
        VpnXrayHevChainTest.stopEverything()
        val quietMs = 3_000L
        val deadline = SystemClock.elapsedRealtime() + 30_000L
        var changedAt = SystemClock.elapsedRealtime()
        var previous = tunnelBusy()
        while (SystemClock.elapsedRealtime() < deadline) {
            val current = tunnelBusy()
            if (current != previous) {
                previous = current
                changedAt = SystemClock.elapsedRealtime()
            } else if (!current && SystemClock.elapsedRealtime() - changedAt >= quietMs) {
                Log.i(TAG, "drain: tunnel quiet for ${SystemClock.elapsedRealtime() - changedAt} ms")
                return
            }
            Thread.sleep(200)
        }
        fail("service teardown did not settle within 30s: busy=$previous")
    }

    private fun tunnelBusy(): Boolean = XrayRuntime.isRunning() || TProxyService.TProxyIsRunning()

    @Test
    fun startStopStartLeavesSingleConsistentInstance() {
        val first = XrayRuntime.start()
        assertTrue("first start failed: ${first.exceptionOrNull()?.message}", first.isSuccess)
        val firstPort = first.getOrThrow()
        assertTrue("port must be reported", firstPort > 0)
        assertEquals("socksPort must match the returned port", firstPort, XrayRuntime.socksPort)

        // Повторный start при живом инстансе не должен создавать второй.
        val second = XrayRuntime.start()
        assertTrue("a second start while running must be rejected", second.isFailure)
        assertEquals("failed second start must not change the port", firstPort, XrayRuntime.socksPort)

        XrayRuntime.stop()
        assertTrue("Xray must be stopped", !XrayRuntime.isRunning())
        assertEquals("port must be cleared on stop", 0, XrayRuntime.socksPort)

        val third = XrayRuntime.start()
        assertTrue("start after stop failed: ${third.exceptionOrNull()?.message}", third.isSuccess)
        val thirdPort = third.getOrThrow()
        assertEquals("port must be consistent after restart", thirdPort, XrayRuntime.socksPort)

        // SOCKS5 должен реально работать после start -> stop -> start.
        assertSocksConnect(thirdPort)
    }

    @Test
    fun repeatedRestartKeepsExactlyOneRunningInstance() {
        val initial = XrayRuntime.start()
        assertTrue("initial start failed: ${initial.exceptionOrNull()?.message}", initial.isSuccess)

        // Handover может потребовать несколько перезапусков подряд (Mobile -> Wi-Fi -> Mobile).
        repeat(3) { attempt ->
            val restarted = XrayRuntime.restart()
            assertTrue(
                "restart #$attempt failed: ${restarted.exceptionOrNull()?.message}",
                restarted.isSuccess,
            )
            val port = restarted.getOrThrow()
            assertEquals("socksPort must match after restart #$attempt", port, XrayRuntime.socksPort)
            assertTrue("Xray must be running after restart #$attempt", XrayRuntime.isRunning())
            assertSocksConnect(port)
        }

        // Ровно один инстанс: повторный start не проходит.
        assertTrue("a second instance appeared after restarts", XrayRuntime.start().isFailure)
    }

    @Test

        XrayRuntime.start()
        XrayRuntime.stop()
        XrayRuntime.stop()
        assertTrue("Xray must stay stopped", !XrayRuntime.isRunning())

        val result = XrayRuntime.restart()
        assertTrue("restart from a stopped state failed: ${result.exceptionOrNull()?.message}", result.isSuccess)
        assertTrue("Xray must be running", XrayRuntime.isRunning())

    /** SOCKS5 CONNECT до CONTROL_HOST:80 — минимальная проверка, что инстанс живой. */
    private fun assertSocksConnect(port: Int) {
        assertNotEquals("socks port must not be zero", 0, port)
        val socket = socksConnect(port, CONTROL_HOST, 80, 20_000)
        socket.use { assertTrue("socket must be connected", it.isConnected) }
        Log.i(TAG, "SOCKS5 CONNECT via 127.0.0.1:$port -> $CONTROL_HOST:80 ok")
    }

    private fun socksConnect(port: Int, host: String, remotePort: Int, timeoutMs: Int): Socket {
        val socket = Socket()
        socket.connect(InetSocketAddress(InetAddress.getByName("127.0.0.1"), port), timeoutMs)
        socket.soTimeout = timeoutMs
        val out: OutputStream = socket.getOutputStream()
        val input: InputStream = socket.getInputStream()

        out.write(byteArrayOf(0x05, 0x01, 0x00))
        out.flush()
        val greeting = ByteArray(2)
        readFully(input, greeting, "greeting")
        check(greeting[0].toInt() and 0xFF == 0x05) { "bad SOCKS version" }
        check(greeting[1].toInt() and 0xFF == 0x00) { "SOCKS5 no-auth rejected" }

        val hostBytes = host.toByteArray(Charsets.US_ASCII)
        val request = ByteArrayOutputStream()
        request.write(0x05)
        request.write(0x01)
        request.write(0x00)
        request.write(0x03)
        request.write(hostBytes.size)
        request.write(hostBytes)
        request.write((remotePort ushr 8) and 0xFF)
        request.write(remotePort and 0xFF)
        out.write(request.toByteArray())
        out.flush()

        val head = ByteArray(4)
        readFully(input, head, "reply")
        val reply = head[1].toInt() and 0xFF
        check(reply == 0x00) { "SOCKS5 CONNECT failed, REP=$reply" }

        val atyp = head[3].toInt() and 0xFF
        val addressLength = when (atyp) {
            0x01 -> 4
            0x04 -> 16
            0x03 -> {
                val len = ByteArray(1)
                readFully(input, len, "domain length")
                len[0].toInt() and 0xFF
            }
            else -> throw IllegalStateException("unknown SOCKS ATYP $atyp")
        }
        readFully(input, ByteArray(addressLength), "bound address")
        readFully(input, ByteArray(2), "bound port")
        return socket
    }

    private fun readFully(input: InputStream, buffer: ByteArray, what: String) {
        var offset = 0
        while (offset < buffer.size) {
            val read = input.read(buffer, offset, buffer.size - offset)
            if (read == -1) throw IllegalStateException("unexpected EOF while reading $what")
            offset += read
        }
    }

    private companion object {
        const val TAG = "XrayHandoverTest"
        const val CONTROL_HOST = "cp.cloudflare.com"
    }
}

package com.kaitag.vpn

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.util.Log

/**
 * TEST-ONLY control channel (debug builds). Allows the instrumentation test (same UID,
 * but a separate process) to ask the app process that owns the VPN consent to start or
 * stop the tunnel.
 *
 * Lives in the `debug` source set and is declared only in the debug manifest, so the
 * release APK/AAB contains no test controls at all.
 *
 * The receiver is declared non-exported in the manifest, so it is reachable only
 * through an explicit intent addressed to this component — other applications
 * cannot start or stop the user's VPN.
 */
class VpnTestControlReceiver : BroadcastReceiver() {

    override fun onReceive(context: Context, intent: Intent) {
        // Non-exported receiver: only our own UID may drive it. Checked explicitly so
        // the protection survives a future manifest change.
        if (intent.component?.packageName != context.packageName) {
            Log.w(TAG, "ignored broadcast for foreign component: ${intent.component}")
            return
        }
        val action = intent.action ?: return
        val serviceIntent = Intent(context, KaitagVpnService::class.java)
        when (action) {
            ACTION_START -> {
                serviceIntent.action = KaitagVpnService.ACTION_CONNECT
                Log.i(TAG, "test-control: $action -> starting service in uid ${android.os.Process.myUid()}")
                context.startForegroundService(serviceIntent)
            }
            ACTION_STOP -> {
                serviceIntent.action = KaitagVpnService.ACTION_DISCONNECT
                Log.i(TAG, "test-control: $action -> stopping service in uid ${android.os.Process.myUid()}")
                // STOP never promotes to foreground, so plain startService is correct here:
                // startForegroundService would demand a startForeground() call and crash.
                context.startService(serviceIntent)
            }
            else -> return
        }
    }

    companion object {
        private const val TAG = "VpnTestControl"

        const val ACTION_START = "com.kaitag.vpn.test.START"
        const val ACTION_STOP = "com.kaitag.vpn.test.STOP"
    }
}

===== app\src\main\java\com\kaitag\vpn\HandoverRetryPolicy.kt =====
package com.kaitag.vpn

private const val DEFAULT_MAX_ATTEMPTS = 4
private const val DEFAULT_BASE_DELAY_MS = 1000L

/**
 * Политика повторных попыток перезапуска датапаса при handover.
 *
 * Каждая неудачная попытка удваивает задержку ([retryDelayMs]), а общее число
 * попыток ограничено [maxAttempts]. Чистая JVM-логика, покрыта unit-тестами.
 */
class HandoverRetryPolicy(
    val maxAttempts: Int = DEFAULT_MAX_ATTEMPTS,
    val baseDelayMs: Long = DEFAULT_BASE_DELAY_MS,
) {

    /**
     * Задержка перед повтором после [failedAttempt]-й неудачи.
     * 1-я → baseDelayMs, 2-я → 2×baseDelayMs, 3-я → 4×baseDelayMs и т.д.
     */
    fun retryDelayMs(failedAttempt: Int): Long {
        require(failedAttempt >= 1) { "failedAttempt must be >= 1, got $failedAttempt" }
        return baseDelayMs * (1L shl (failedAttempt - 1))
    }

    /** Можно ли повторить после [failedAttempt]-й неудачи (последняя не повторяется). */
    fun canRetry(failedAttempt: Int): Boolean {
        require(failedAttempt >= 1) { "failedAttempt must be >= 1, got $failedAttempt" }
        return failedAttempt in 1 until maxAttempts
    }

    init {
        require(maxAttempts >= 1) { "maxAttempts must be >= 1, got $maxAttempts" }
        require(baseDelayMs >= 0) { "baseDelayMs must be >= 0, got $baseDelayMs" }
    }
}

package com.kaitag.vpn

import android.app.Notification
import android.app.NotificationChannel

import androidx.core.app.ServiceCompat
import hev.htproxy.TProxyService
import java.io.File

class KaitagVpnService : VpnService() {

    private var tunFd: ParcelFileDescriptor? = null
    private var connectivityManager: ConnectivityManager? = null
    private var networkCallback: ConnectivityManager.NetworkCallback? = null

    /** Сериализует все операции с туннелем (старт/стоп/handover). */
    private val tunnelLock = Any()

    /** Один рабочий поток для перезапуска Xray, чтобы не блокировать main/binder. */
    private var workerHandler: Handler? = null
    private var handoverHandler: Handler? = null

    private val handover = NetworkHandoverDecider()

    /** Гарантирует ровно один перезапуск: compareAndSet вместо проверки-и-записи. */
    private val restartGuard = java.util.concurrent.atomic.AtomicBoolean(false)

    /** Политика повторных попыток перезапуска датапаса при handover. */
    private val handoverRetry = HandoverRetryPolicy()

    @Volatile
    private var stopping = false

@Volatile
    private var destroyed = false

    override fun onCreate() {
        super.onCreate()
        // Потоки создаются один раз, чтобы не было гонки ленивой инициализации.
        val handoverThread = HandlerThread("kaitag-handover").also { it.start() }
        handoverHandler = Handler(handoverThread.looper)
        val workerThread = HandlerThread("kaitag-network-worker").also { it.start() }
        workerHandler = Handler(workerThread.looper)

        // В release не логируем внутренности Xray (debug-лог раскрывает направления
        // трафика); в debug-сборках оставляем диагностику для тестов и разработки.
        val debuggable = applicationInfo.flags and ApplicationInfo.FLAG_DEBUGGABLE != 0
        XrayRuntime.logLevel = if (debuggable) "debug" else "warning"
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        // Запрос состояния обрабатывается до foreground-промоушена: UI спрашивает факт,
        // а не запускает сервис, и уведомление при этом не нужно.
        if (intent != null && ACTION_QUERY_STATE == intent.action) {
            broadcastState(isRunning = tunFd != null)
            if (tunFd == null) {
                // Ничего не запущено: пустой сервис не должен оставаться started,
                // иначе он висит фоном, а после убийства процесса система ещё и
                // пересоздаст его (sticky) без единой причины.
                stopSelf(startId)
                return START_NOT_STICKY
            }
            // Туннель живёт: не меняем sticky-флаг, выставленный ACTION_CONNECT.
            return START_STICKY
        }

        // Промоушен до всех остальных веток: запрос мог прийти через
        // startForegroundService(), а платформа убивает процесс, если соответствующий
        // startForeground() не вызван вовремя — в том числе на путях DISCONNECT и
        // на путях, где запрос в итоге игнорируется.
        if (intent != null) {
            startForegroundNotification()
        }

        if (intent != null && ACTION_DISCONNECT == intent.action) {
            stopTunnel()
            stopSelf()
            return START_NOT_STICKY
        }

        // VPN поднимается ТОЛЬКО по явной команде ACTION_CONNECT. После убийства процесса
        // система пересоздаёт сервис с intent == null (START_STICKY), и одного лишь
        // tunFd == null недостаточно, чтобы поднять туннель: пользователь об этом не просил.
        if (intent == null || ACTION_CONNECT != intent.action) {
            Log.i(TAG, "старт без ACTION_CONNECT проигнорирован: intent=$intent action=${intent?.action}")
            // Публикуем факт: слушатель видит, что сервис обработал запрос и туннель не поднят.
            broadcastState(isRunning = tunFd != null)
            if (tunFd == null) {
                // Ничего не запущено: сервис не должен висеть пустым, а START_STICKY
                // не должен пересоздавать его снова без команды. stopSelf(startId)
                // не трогает более поздний START, поэтому параллельный ACTION_CONNECT
                // (если он уже в очереди) отрабатывает нормально.
                stopSelf(startId)
            }
            return START_NOT_STICKY
        }

        if (tunFd == null) {
            startTunnel()
        } else {
            // Already up: re-publish state so a late listener (the UI after a restart) can observe it.
            broadcastState(isRunning = true)
        }
        return START_STICKY
    }

    override fun onRevoke() {
        stopTunnel()
        super.onRevoke()
    }

    override fun onDestroy() {
        destroyed = true
        stopTunnel()
        // Колбэк снимается вместе с туннелем, рабочие потоки останавливаем здесь,
        // чтобы сервис не держал их после уничтожения.
        handoverHandler?.removeCallbacksAndMessages(null)
        workerHandler?.removeCallbacksAndMessages(null)
        workerHandler?.looper?.quitSafely()
        handoverHandler?.looper?.quitSafely()
        workerHandler = null
        handoverHandler = null
        super.onDestroy()
    }

    /**
     * Поднимает туннель. SOCKS-порт Xray выбирается самим XrayRuntime, поэтому
     * параметры сюда не передаются: hev получает реальный порт из [XrayRuntime.start].
     */
    fun startTunnel() {
        synchronized(tunnelLock) {
            if (tunFd != null) {
                return
            }
            stopping = false

            // Promote to foreground first: establish() + Xray startup can exceed the
            // ForegroundServiceDidNotStartInTime window if notification comes later.
            startForegroundNotification()

            // Xray first: hev-socks5-tunnel needs a live loopback SOCKS5 endpoint.
            val xrayStarted = XrayRuntime.start()
            if (xrayStarted.isFailure) {
                cleanupFailedStart()
                failWith("Xray не запустился: ${xrayStarted.exceptionOrNull()?.message}")
                return
            }
            val xraySocksPort = xrayStarted.getOrThrow()

            Log.i(
                TAG,
                "establish() вызывается, prepare=${VpnService.prepare(this)?.javaClass?.simpleName ?: "null (согласие есть)"}",
            )
            val descriptor = try {
                Builder()
                    .setSession(SESSION_NAME)
                    .setBlocking(false)
                    .setMtu(TUNNEL_MTU)
                    .addAddress(TUNNEL_IPV4_ADDRESS, TUNNEL_IPV4_PREFIX)
                    .addRoute("0.0.0.0", 0)
                    .addDnsServer(TUNNEL_DNS_ADDRESS)
                    .addDisallowedApplication(packageName)
                    .establish()
            } catch (e: Exception) {
                Log.e(TAG, "establish() бросил исключение", e)
                cleanupFailedStart()
                failWith("исключение establish(): ${e.message}")
                return
            }
            Log.i(TAG, "establish() вернул ${if (descriptor == null) "null" else "fd=${descriptor.fd}"}")

            if (descriptor == null) {
                cleanupFailedStart()
                failWith("не удалось создать TUN-интерфейс")
                return
            }
            tunFd = descriptor

            if (!startHev(descriptor, xraySocksPort, tearDownOnFailure = true)) {
                return
            }

            broadcastState(isRunning = true)

            // Следим за нижележащей сетью, чтобы туннель выживал при переходе Wi-Fi <-> мобильная.
            startNetworkTracking()
        }
    }

    /**
     * Запускает hev на готовом TUN.
     *
     * @param tearDownOnFailure при первичном запуске сбой означает, что туннеля нет и
     *        нужно убрать за собой; при handover туннель уже работал, поэтому его
     *        сохраняем — приложения продолжат работать, а перезапуск можно повторить.
     */
    private fun startHev(
        descriptor: ParcelFileDescriptor,
        xraySocksPort: Int,
        tearDownOnFailure: Boolean,
    ): Boolean {
        val upstreamAddress = XRAY_SOCKS_ADDRESS
        val configFile = File(cacheDir, CONFIG_FILE_NAME)
        try {
            configFile.writeText(buildConfig(upstreamAddress, xraySocksPort))
        } catch (e: IOException) {
            Log.e(TAG, "ошибка записи конфигурации hev", e)
            if (tearDownOnFailure) {
                closeTun()
                XrayRuntime.stop()
                ServiceCompat.stopForeground(this, ServiceCompat.STOP_FOREGROUND_REMOVE)
                failWith("ошибка записи конфигурации: ${e.message}")
            } else {
                Log.w(TAG, "startHev: конфиг не записан на recover/restart-пути, туннель сохранён")
            }
            return false
        }

        if ((applicationInfo.flags and ApplicationInfo.FLAG_DEBUGGABLE) != 0) {
            Log.i(TAG, "hev конфиг:\n${configFile.readText()}")
        }
        Log.i(TAG, "HEV starting: fd=${descriptor.fd} upstream=$upstreamAddress:$xraySocksPort")
        if (!TProxyService.TProxyStartService(configFile.absolutePath, descriptor.fd)) {
            Log.e(TAG, "TProxyStartService вернул false")
            if (tearDownOnFailure) {
                closeTun()
                XrayRuntime.stop()
                ServiceCompat.stopForeground(this, ServiceCompat.STOP_FOREGROUND_REMOVE)
                failWith("hev-socks5-tunnel не запустился (SOCKS5 $upstreamAddress:$xraySocksPort недоступен)")
            } else {
                Log.w(TAG, "startHev: HEV не стартовал на recover/restart-пути, туннель сохранён")
            }
            return false
        }
        Log.i(TAG, "HEV started, upstream=$upstreamAddress:$xraySocksPort")
        return true
    }

    /** Убирает всё, что успело подняться до неудачного запуска. */
    private fun cleanupFailedStart() {
        TProxyService.TProxyStopService()
        XrayRuntime.stop()
        closeTun()
        stopNetworkTracking()
        ServiceCompat.stopForeground(this, ServiceCompat.STOP_FOREGROUND_REMOVE)
    }

    /** TEST-ONLY: lets the instrumentation test observe tunnel state without extra UI. */
    private fun broadcastState(isRunning: Boolean) {
        sendBroadcast(
            Intent(ACTION_TUNNEL_STATE)
                .setPackage(packageName)
                .putExtra(EXTRA_TUNNEL_RUNNING, isRunning)
                .putExtra(EXTRA_XRAY_SOCKS_PORT, XrayRuntime.socksPort)
                .putExtra(EXTRA_TUNNEL_UPSTREAM, "$XRAY_SOCKS_ADDRESS:${XrayRuntime.socksPort}"),
        )
    }

    private fun failWith(reason: String) {
        Log.e(TAG, "ОТКАЗ: $reason")
        sendBroadcast(
            Intent(ACTION_TUNNEL_ERROR)
                .setPackage(packageName)
                .putExtra(EXTRA_ERROR_MESSAGE, reason),
        )
        stopSelf()
    }

    fun stopTunnel() {
        synchronized(tunnelLock) {
            stopping = true
            cancelPendingHandover()
            // Everything is torn down regardless of how far the startup got: stopTunnel()
            // may be called when the TUN never came up but Xray or hev already started.
            if (TProxyService.TProxyIsRunning()) {
                Log.i(TAG, "HEV stopping")
                TProxyService.TProxyStopService()
                Log.i(TAG, "HEV stopped")
            }
            if (XrayRuntime.isRunning()) {
                Log.i(TAG, "Xray stopping")
                XrayRuntime.stop()
                Log.i(TAG, "Xray stopped")
            }
            stopNetworkTracking()
            closeTun()
            ServiceCompat.stopForeground(this, ServiceCompat.STOP_FOREGROUND_REMOVE)
            // Публикуем состояние всегда, а не только если раньше было running=true:
            // иначе DISCONNECT по мёртвому сервису оставляет UI в состоянии
            // «Отключение…» без единого ответа.
            broadcastState(isRunning = false)
            handover.reset()
        }
    }

    private fun closeTun() {
        try {
            tunFd?.close()
        } catch (e: IOException) {
        }
        tunFd = null
    }

    // ---------------------------------------------------------------- network tracking

    private fun startNetworkTracking() {
        if (networkCallback != null) {
            return
        }
        val manager = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val callback = object : ConnectivityManager.NetworkCallback() {
            override fun onAvailable(network: Network) {
                // Network из callback'а уходит дальше в evaluateNetworks(): снимок и
                // setUnderlyingNetworks() считаются от неё, а не от activeNetwork,
                // которая при быстром Mobile -> Wi-Fi -> Mobile уже успела смениться.
                Log.i(TAG, "network available: ${describe(network)}")
                evaluateNetworks("onAvailable", network)
            }

            override fun onCapabilitiesChanged(
                network: Network,
                networkCapabilities: NetworkCapabilities,
            ) {
                // Не перезапускаем на каждое изменение capabilities: только пересчитываем
                // состояние и, если нужно, планируем одну отложенную проверку.
                Log.i(
                    TAG,
                    "network capabilities: ${describe(network, networkCapabilities)} " +
                        "internet=${networkCapabilities.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)} " +
                        "validated=${networkCapabilities.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED)}",
                )
                evaluateNetworks("onCapabilitiesChanged", network)
            }

            override fun onLost(network: Network) {
                // Сеть из callback'а уже недоступна: evaluateNetworks() убедится в этом
                // и только тогда прочитает актуальную default network.
                Log.i(TAG, "network lost: ${describe(network)}")
                evaluateNetworks("onLost", network)
            }
        }
        try {
            manager.registerDefaultNetworkCallback(callback)
        } catch (e: Exception) {
            Log.e(TAG, "не удалось подписаться на смену сети", e)
            return
        }
        connectivityManager = manager
        networkCallback = callback

        // Первичная привязка: только запоминаем сеть, перезапуск не планируем.
        evaluateNetworks("initial")
    }

    private fun stopNetworkTracking() {
        val manager = connectivityManager
        val callback = networkCallback
        if (manager != null && callback != null) {
            try {
                manager.unregisterNetworkCallback(callback)
            } catch (e: Exception) {
                Log.w(TAG, "unregisterNetworkCallback: ${e.message}")
            }
        }
        connectivityManager = null
        networkCallback = null
        handover.reset()
    }

    /**
     * Сеть для снимка и для [setUnderlyingNetworks]: объект из callback'а, пока он ещё
     * отслеживается системой. Если сети уже нет (типовой случай — onLost) либо
     * callback'а не было (первичная привязка, проверка после debounce/перезапуска),
     * читается актуальная default network — это единственное место перечитывания.
     */
    private fun resolveNetwork(source: Network?): Pair<Network, NetworkCapabilities>? {
        val manager = connectivityManager ?: return null
        if (source != null) {
            val sourceCaps = manager.getNetworkCapabilities(source)
            if (sourceCaps != null) {
                if (!sourceCaps.hasTransport(NetworkCapabilities.TRANSPORT_VPN)) {
                    return source to sourceCaps
                }
                // Свой VPN-интерфейс кандидатом на handover быть не может:
                // перечитываем актуальную нижележащую сеть.
                Log.i(TAG, "callback network is our own VPN; reading underlying default network")
            } else {
                Log.i(TAG, "callback network no longer tracked, reading default network: ${describe(source)}")
            }
        }
        val candidate = activeUnderlyingNetwork(manager) ?: return null
        val caps = manager.getNetworkCapabilities(candidate) ?: return null
        return candidate to caps
    }

    /**
     * Актуальная сеть под нашим VPN-туннелем: сам VPN-интерфейс (TRANSPORT_VPN)
     * исключается, иначе и decider, и setUnderlyingNetworks() получили бы наш же
     * туннель вместо физической сети.
     */
    private fun activeUnderlyingNetwork(manager: ConnectivityManager): Network? {
        val active = manager.activeNetwork
        val candidates = if (active != null) {
            sequenceOf(active) + manager.allNetworks.asSequence()
        } else {
            manager.allNetworks.asSequence()
        }
        for (network in candidates.distinct()) {
            val caps = manager.getNetworkCapabilities(network) ?: continue
            if (caps.hasTransport(NetworkCapabilities.TRANSPORT_VPN)) continue
            if (!caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)) continue
            return network
        }
        return null
    }

    private fun snapshotOf(
        network: Network,
        caps: NetworkCapabilities,
    ): NetworkHandoverDecider.Snapshot = NetworkHandoverDecider.Snapshot(
        id = networkId(network),
        type = describe(network, caps),
        hasInternet = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET),
        validated = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED),
    )

    /**
     * Единственное место, где принимается решение о handover.
     *
     * @param source Network из callback'а — она же используется и для снимка, и для
     *        setUnderlyingNetworks(). null означает «callback'а не было»: тогда берётся
     *        актуальная default network.
     */
    private fun evaluateNetworks(trigger: String, source: Network? = null) {
        if (tunFd == null || stopping || destroyed) {
            return
        }
        val resolved = resolveNetwork(source)
        if (resolved == null) {
            // Ни callback-сети, ни default network нет: decider на null-снимке всегда
            // отвечает Ignore и своё состояние не меняет.
            return
        }
        val (network, caps) = resolved
        val snapshot = snapshotOf(network, caps)
        when (val decision = handover.onDefaultNetwork(snapshot)) {
            is NetworkHandoverDecider.Decision.Ignore -> Unit
            is NetworkHandoverDecider.Decision.Adopt -> {
                Log.i(TAG, "initial underlying network: id=${decision.networkId} type=${decision.type} (trigger=$trigger)")
                applyUnderlyingNetworks(network, decision.type)
            }
            is NetworkHandoverDecider.Decision.CancelPending -> {
                Log.i(TAG, "pending handover cancelled, back on id=${decision.networkId} type=${decision.type}")
            }
            is NetworkHandoverDecider.Decision.ScheduleRestart -> {
                Log.i(
                    TAG,
                    "handover detected (trigger=$trigger): ${decision.fromType ?: "none"} -> ${decision.toType} " +
                        "id=${decision.networkId}; restart scheduled in ${HANDOVER_DELAY_MS}ms",
                )
                scheduleHandoverCheck()
            }
        }
    }

    private fun scheduleHandoverCheck() {
        cancelPendingHandover()
        handoverHandler?.postDelayed({ confirmHandover() }, HANDOVER_DELAY_MS)
    }

    private fun cancelPendingHandover() {
        handoverHandler?.removeCallbacksAndMessages(null)
    }

    /**
     * Истекла задержка стабилизации: заново читаем актуальную сеть и, если она
     * validated, выполняем ровно один перезапуск Xray. Один и тот же объект Network
     * используется и для снимка, и для setUnderlyingNetworks().
     */
    private fun confirmHandover() {
        if (tunFd == null || stopping) {
            handover.onRestartFailed()
            return
        }
        val resolved = resolveNetwork(source = null)
        if (resolved == null) {
            // Окно подтверждения истекло, а usable-сети нет: снимаем ожидание,
            // иначе handover зависнет — следующий подходящий callback должен
            // запланировать debounce заново, а не упереться в застрявший pending.
            handover.onRestartFailed()
            Log.i(TAG, "handover check: no usable default network; pending cleared")
            return
        }
        val (network, caps) = resolved
        val snapshot = snapshotOf(network, caps)
        when (val decision = handover.onDefaultNetwork(snapshot, maintenance = true)) {
            is NetworkHandoverDecider.Decision.ScheduleRestart -> {
                Log.i(TAG, "handover confirmed -> restart started: ${decision.fromType ?: "none"} -> ${decision.toType}")
                applyUnderlyingNetworks(network, decision.toType)
                restartDataPath(decision.networkId, decision.toType)
            }
            is NetworkHandoverDecider.Decision.Ignore ->
                Log.i(TAG, "handover check: network unchanged, no restart")
            is NetworkHandoverDecider.Decision.CancelPending ->
                Log.i(TAG, "handover check: back on previous network, no restart")
            is NetworkHandoverDecider.Decision.Adopt -> {
                applyUnderlyingNetworks(network, decision.type)
            }
        }
    }

    private fun applyUnderlyingNetworks(network: Network, type: String) {
        // setUnderlyingNetworks имеет смысл только при живом VPN-интерфейсе.
        if (tunFd == null) {
            Log.w(TAG, "skip setUnderlyingNetworks: VPN interface is gone")
            return
        }
        // Тот же объект Network, что и в снимке: activeNetwork заново не читаем, иначе
        // быстрое переключение Mobile -> Wi-Fi -> Mobile даёт рассогласование
        // «решение принято по одной сети, а применено к другой».
        val result = setUnderlyingNetworks(arrayOf(network))
        Log.i(TAG, "setUnderlyingNetworks(id=${networkId(network)} type=$type) -> $result")
    }

    /**
     * Смена канала Mobile <-> Wi-Fi. TUN и VPN-интерфейс НЕ трогаем: пересоздание
     * интерфейса одновременно с переключением физической сети даёт лишнюю гонку.
     * Перезапускаем только внешнее соединение — Xray (и hev, если сменился SOCKS-порт).
     */
    private fun restartDataPath(networkId: Long, type: String) {
        if (!restartGuard.compareAndSet(false, true)) {
            Log.i(TAG, "restart already in flight; the ongoing restart will re-check the network")
            return
        }
        val handler = workerHandler
        if (handler == null) {
            Log.e(TAG, "restart worker is gone; skipping")
            restartGuard.set(false)
            return
        }
        handler.post { attemptDataPathRecovery(networkId, type, attempt = 1, handler = handler) }
    }

    /**
     * Одна попытка + расписание повторов. Guard удерживается, пока идут попытки:
     * параллельный restart невозможен. Первая попытка — полный перезапуск Xray/hev,
     * повторы — точечное восстановление только упавшего компонента.
     */
    private fun attemptDataPathRecovery(networkId: Long, type: String, attempt: Int, handler: Handler) {
        val ok = if (attempt == 1) performDataPathRestart() else recoverDataPath()
        if (ok) {
            try {
                handover.onRestartCompleted(networkId, type)
                broadcastState(isRunning = true)
                Log.i(TAG, "restart completed on id=$networkId type=$type (attempt $attempt)")
            } finally {
                restartGuard.set(false)
            }
            evaluateNetworks("afterRestart")
            return
        }
        if (!handoverRetry.canRetry(attempt)) {
            Log.e(TAG, "restart failed on id=$networkId type=$type after $attempt attempt(s); giving up")
            try {
                handover.onRestartFailed()
            } finally {
                restartGuard.set(false)
            }
            evaluateNetworks("afterRestart")
            return
        }
        val delayMs = handoverRetry.retryDelayMs(attempt)
        Log.e(TAG, "restart failed on id=$networkId type=$type (attempt $attempt); retry in ${delayMs}ms")
        handler.postDelayed({ attemptDataPathRecovery(networkId, type, attempt + 1, handler) }, delayMs)
    }

    /**
     * Повторная попытка после первого сбоя: TUN и VPN-интерфейс уже работали и были
     * сохранены, поэтому перезапускаем только упавший компонент (Xray и/или hev).
     */
    private fun recoverDataPath(): Boolean {
        return try {
            synchronized(tunnelLock) {
                if (tunFd == null || stopping) {
                    Log.i(TAG, "recover skipped: tunnel is down")
                    return@synchronized false
                }
                val descriptor = tunFd ?: return@synchronized false
                if (!XrayRuntime.isRunning()) {
                    Log.w(TAG, "recover: Xray is down, starting it again")
                    val started = XrayRuntime.restart()
                    if (started.isFailure) {
                        Log.e(TAG, "recover failed: Xray did not start: ${started.exceptionOrNull()?.message}")
                        return@synchronized false
                    }
                    Log.i(TAG, "recover: Xray restarted on ${XrayRuntime.socksPort}")
                }
                if (!TProxyService.TProxyIsRunning()) {
                    Log.w(TAG, "recover: HEV is down, starting it on the current SOCKS port")
                    if (!startHev(descriptor, XrayRuntime.socksPort, tearDownOnFailure = false)) {
                        Log.e(TAG, "recover failed: HEV did not start")
                        return@synchronized false
                    }
                    Log.i(TAG, "recover: HEV started")
                }
                val xrayUp = XrayRuntime.isRunning()
                val hevUp = TProxyService.TProxyIsRunning()
                if (!xrayUp || !hevUp) {
                    Log.e(TAG, "recover failed: xrayRunning=$xrayUp hevRunning=$hevUp")
                    return@synchronized false
                }
                true
            }
        } catch (e: Exception) {
            Log.e(TAG, "recover failed with exception", e)
            false
        }
    }

    /**
     * Перезапускает только внешнее соединение. TUN и VPN-интерфейс сохраняются:
     * пересоздание интерфейса одновременно с переключением физической сети даёт
     * лишнюю гонку и может уронить интернет.
     */
    private fun performDataPathRestart(): Boolean {
        return try {
            synchronized(tunnelLock) {
                if (tunFd == null || stopping) {
                    Log.i(TAG, "restart skipped: tunnel is down")
                    return@synchronized false
                }
                val descriptor = tunFd ?: return@synchronized false
                val oldPort = XrayRuntime.socksPort

                Log.i(TAG, "restart started: stopping HEV and Xray, TUN kept")
                TProxyService.TProxyStopService()
                Log.i(TAG, "HEV stopped")

                val started = XrayRuntime.restart()
                if (started.isFailure) {
                    Log.e(TAG, "restart failed: Xray did not start: ${started.exceptionOrNull()?.message}")
                    return@synchronized false
                }
                val newPort = started.getOrThrow()
                Log.i(TAG, "Xray started on SOCKS5 127.0.0.1:$newPort (was $oldPort)")

                if (newPort != oldPort || !TProxyService.TProxyIsRunning()) {
                    Log.i(TAG, "restart: HEV restart needed (port changed or HEV down)")
                    // Туннель уже работал: при сбое hev его не уничтожаем.
                    if (!startHev(descriptor, newPort, tearDownOnFailure = false)) {
                        Log.e(TAG, "restart failed: HEV did not start")
                        return@synchronized false
                    }
                    Log.i(TAG, "HEV started")
                }

                val xrayUp = XrayRuntime.isRunning()
                val hevUp = TProxyService.TProxyIsRunning()
                if (!xrayUp || !hevUp) {
                    Log.e(TAG, "restart failed: xrayRunning=$xrayUp hevRunning=$hevUp")
                    return@synchronized false
                }
                true
            }
        } catch (e: Exception) {
            Log.e(TAG, "restart failed with exception", e)
            false
        }
    }

    /** Стабильный идентификатор сети для сравнения. */
    private fun networkId(network: Network): Long = network.networkHandle

    private fun describe(network: Network): String {
        val manager = connectivityManager
        val caps = manager?.getNetworkCapabilities(network)
        return describe(network, caps)
    }

    private fun describe(network: Network, caps: NetworkCapabilities?): String {
        val transports = buildString {
            if (caps == null) {
                append("unknown")
            } else {
                if (caps.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)) append("wifi")
                if (caps.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR)) {
                    if (isNotEmpty()) append("|")
                    append("cellular")
                }
                if (caps.hasTransport(NetworkCapabilities.TRANSPORT_VPN)) {
                    if (isNotEmpty()) append("|")
                    append("vpn")
                }
                if (caps.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET)) {
                    if (isNotEmpty()) append("|")
                    append("ethernet")
                }
                if (isEmpty()) append("other")
            }
        }
        return "$transports#${network.networkHandle}"
    }

    private fun buildConfig(socksAddress: String, socksPort: Int): String = buildString {
        append("misc:\n")
        append("  log-level: 'warn'\n")
        append("tunnel:\n")
        append("  mtu: $TUNNEL_MTU\n")
        append("  icmp: 'reply'\n")
        append("socks5:\n")
        append("  address: '$socksAddress'\n")
        append("  port: $socksPort\n")
        append("  udp: 'udp'\n")
    }

    private fun startForegroundNotification() {
        val manager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val channel = NotificationChannel(
            NOTIFICATION_CHANNEL_ID,
            getString(R.string.app_name),
            NotificationManager.IMPORTANCE_DEFAULT,
        )

        val contentIntent = PendingIntent.getActivity(

            0,
            Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE,
        )
        val notification: Notification = NotificationCompat.Builder(this, NOTIFICATION_CHANNEL_ID)
            .setContentTitle(getString(R.string.app_name))
            .setContentText(getString(R.string.vpn_notification_text))
            .setSmallIcon(android.R.drawable.sym_def_app_icon)
            .setContentIntent(contentIntent)
            .setOngoing(true)
            .build()

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            ServiceCompat.startForeground(
                this,
                NOTIFICATION_ID,
                notification,
                ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE,
            )
        } else {
            ServiceCompat.startForeground(this, NOTIFICATION_ID, notification, 0)
        }
    }

     * до запроса, поэтому ни повторный запуск, ни повторные подключения не спамят.

    private fun maybeRequestNotificationPermission() {

===== app\src\main\java\com\kaitag\vpn\NetworkHandoverDecider.kt =====
package com.kaitag.vpn

/**
 * Чистая логика решения о смене нижележащей сети. Без зависимостей от Android,
 * поэтому покрывается обычными JVM unit-тестами.
 *
 * Задача: во время Mobile -> Wi-Fi Android присылает несколько callback'ов и
 * промежуточные состояния. Перезапускать VPN на каждом из них нельзя — нужен
 * ровно один перезапуск и только когда новый default network реально пригоден.
 */
class NetworkHandoverDecider {

    /** Снимок состояния сети, полученный из callback'а или из ConnectivityManager. */
    data class Snapshot(
        val id: Long,
        val type: String,
        val hasInternet: Boolean,
        val validated: Boolean,
    ) {
        val usable: Boolean get() = hasInternet && validated
    }

    sealed class Decision {
        /** Ничего не делаем. */
        object Ignore : Decision()

        /** Первичная привязка: запомнить сеть, перезапуск не нужен. */
        data class Adopt(val networkId: Long, val type: String) : Decision()

        /** Сеть реально сменилась: запустить debounce и затем один перезапуск. */
        data class ScheduleRestart(val networkId: Long, val fromType: String?, val toType: String) : Decision()

        /** Во время ожидания сеть изменилась: перезапуск не нужен, ждём нового события. */
        data class CancelPending(val networkId: Long, val type: String) : Decision()
    }

    private var currentNetworkId: Long = NONE
    private var currentType: String? = null
    private var pendingNetworkId: Long = NONE

    val pending: Boolean get() = pendingNetworkId != NONE
    val hasCurrent: Boolean get() = currentNetworkId != NONE

    /**
     * @param maintenance true, если вызов сделан по истечении debounce-задержки.
     *        В этом режиме подтверждается переключение на уже ожидаемую сеть.
     */
    @Synchronized
    fun onDefaultNetwork(snapshot: Snapshot?, maintenance: Boolean = false): Decision {
        if (snapshot == null || !snapshot.usable) {
            // Промежуточное состояние (Wi-Fi появился, но ещё не validated) —
            // не трогаем рабочую сеть.
            if (maintenance && pendingNetworkId != NONE) {
                // Окно подтверждения истекло, а сеть так и не стала пригодной:
                // снимаем ожидание, иначе handover зависнет — следующие callbacks
                // этой же сети отвечали бы Ignore и перезапуск не случился бы никогда.
                pendingNetworkId = NONE
            }
            return Decision.Ignore
        }

        if (pendingNetworkId != NONE) {
            // Пока ждали стабилизации, состояние изменилось.
            if (snapshot.id == pendingNetworkId) {
                // Та же новая сеть: задержку не сбрасываем. Перезапуск выполняем
                // только когда истекла сама задержка (maintenance-раунд).
                return if (maintenance) {
                    Decision.ScheduleRestart(snapshot.id, currentType, snapshot.type)
                } else {
                    Decision.Ignore
                }
            }
            if (snapshot.id == currentNetworkId) {
                // Вернулись на прежнюю сеть — переключение отменяем.
                pendingNetworkId = NONE
                return Decision.CancelPending(snapshot.id, snapshot.type)
            }
            // Уже третья сеть: переключаемся на неё, ожидание снимаем.
            pendingNetworkId = NONE
        } else if (snapshot.id == currentNetworkId) {
            return Decision.Ignore
        }
        if (!hasCurrent && !maintenance) {
            // Первичный запуск VPN: только фиксируем сеть, перезапуск не нужен.
            currentNetworkId = snapshot.id
            currentType = snapshot.type
            pendingNetworkId = NONE
            return Decision.Adopt(snapshot.id, snapshot.type)
        }

        pendingNetworkId = snapshot.id
        return Decision.ScheduleRestart(snapshot.id, currentType, snapshot.type)
    }

    /** Перезапуск успешно завершён — запоминаем сеть, на которой работаем. */
    @Synchronized
    fun onRestartCompleted(networkId: Long, type: String) {
        currentNetworkId = networkId
        currentType = type
        pendingNetworkId = NONE
    }

    /** Перезапуск не удался — текущей остаётся прежняя сеть, ожидание снимаем. */
    @Synchronized
    fun onRestartFailed() {
        pendingNetworkId = NONE
    }

    /** VPN остановлен: сбрасываем состояние, чтобы следующий запуск был «первым». */
    @Synchronized
    fun reset() {
        currentNetworkId = NONE
        currentType = null
        pendingNetworkId = NONE
    }

    private companion object {
        const val NONE = -1L
    }
}

package com.kaitag.vpn

import android.util.Log
import libXray.LibXray
import org.json.JSONArray
import org.json.JSONObject

/**
 * Owns the Xray instance that terminates the loopback SOCKS5 endpoint consumed by
 * hev-socks5-tunnel. Uses only official libXray APIs (apiVersion 3).
 */
object XrayRuntime {

    const val LOG_TAG = "KaitagXray"

    /** Все переходы start/stop последовательны: два start() одновременно невозможны. */
    private val lifecycleLock = Any()

    /**
     * Уровень лога Xray: "debug" в debug-сборках (диагностика трафика), "warning" —
     * в release (никаких секретов и адресов назначения в продакшн-логах).
     * Выставляется сервисом в [KaitagVpnService.onCreate].
     */
    @Volatile
    var logLevel: String = "debug"

    @Volatile
    var socksPort: Int = 0
        private set

    fun start(): Result<Int> = synchronized(lifecycleLock) { startLocked() }

    /**
     * Перезапуск внешнего соединения на новом канале. Старый порт возвращается
     * в набор кандидатов, поэтому hev может продолжить работать без перезапуска.
     *
     * stopXray/runXray обрабатываются в libXray асинхронно: состояние running успевает
     * переключиться в false, но teardown старого инстанса завершается чуть позже и может
     * погасить только что поднятый. Поэтому после stop+start ждём стабилизации, и если
     * свежий инстанс «умер» в первые мгновения — перезапускаем (ограниченное число раз).
     * Гарантия: restart() возвращает успех только при живом устоявшемся инстансе.
     */
    fun restart(): Result<Int> = synchronized(lifecycleLock) {
        Log.i(LOG_TAG, "restart: stopping before start")
        var result: Result<Int> = Result.failure(
            IllegalStateException("restart: истрачены все попытки стабилизации"),
        )
        for (attempt in 1..RESTART_ATTEMPTS) {
            stopLocked()
            // Окно для асинхронного teardown предыдущего инстанса.
            Thread.sleep(RESTART_QUIESCE_MS)
            val started = startLocked()
            if (started.isFailure) {
                Log.w(LOG_TAG, "restart: попытка $attempt не удалась: ${started.exceptionOrNull()?.message}")
                result = started
                continue
            }
            val stable = awaitStableRunning(started.getOrThrow(), RESTART_STABILIZE_MS)
            Log.i(LOG_TAG, "restart: порт ${started.getOrThrow()}, попытка $attempt, stable=$stable")
            if (stable) {
                result = started
                break
            }
            stopLocked()
            result = Result.failure(IllegalStateException("restart: инстанс погас сразу после старта"))
        }
        result
    }

    fun stop() = synchronized(lifecycleLock) { stopLocked() }

    private fun startLocked(): Result<Int> {
        if (isRunningLocked()) {
            return Result.failure(IllegalStateException("Xray уже запущен"))
        }
        val port = pickFreePort().getOrElse {
            return Result.failure(IllegalStateException("не удалось получить свободный порт: ${it.message}"))
        }
        val config = buildVlessConfig(port)
        return invoke("testXray", JSONObject().put("xrayJson", config).toString())
            .map { invoke("runXray", JSONObject().put("xrayJson", config).toString()) }
            .mapCatching {
                check(awaitRunning()) { "Xray не перешёл в состояние running" }
                socksPort = port
                Log.i(LOG_TAG, "Xray запущен, SOCKS5 127.0.0.1:$port")
                port
            }
            .onFailure {
                Log.e(LOG_TAG, "не удалось запустить Xray: ${it.message}")
                invoke("stopXray", "{}")
                resetDNS()
                socksPort = 0
            }
    }

    private fun stopLocked() {
        if (!isRunningLocked()) {
            socksPort = 0
            return
        }
        invoke("stopXray", "{}")
        resetDNS()
        Log.i(LOG_TAG, "Xray остановлен, running=${isRunningLocked()}")
        socksPort = 0
    }

    /** getXrayState обновляется асинхронно, поэтому ждём состояние с коротким таймаутом. */
    private fun awaitRunning(timeoutMs: Long = RUNNING_TIMEOUT_MS): Boolean {
        val deadline = System.currentTimeMillis() + timeoutMs
        while (System.currentTimeMillis() < deadline) {
            if (isRunningLocked()) {
                return true
            }
            Thread.sleep(RUNNING_POLL_MS)
        }
        return isRunningLocked()
    }

    /**
     * Инстанс считается стабильным, если флаг running не мигал ВСЁ окно [windowMs] и на
     * его порту реально слушает SOCKS-сокет: getXrayState может оставаться true за счёт
     * старого (недо-умершего) инстанса, тогда как новый на своём порту не поднялся.
     */
    private fun awaitStableRunning(port: Int, windowMs: Long): Boolean {
        val deadline = System.currentTimeMillis() + windowMs
        while (System.currentTimeMillis() < deadline) {
            if (!isRunningLocked()) {
                return false
            }
            if (socksViable(port)) {
                return true
            }
            Thread.sleep(RUNNING_POLL_MS)
        }
        return isRunningLocked() && socksViable(port)
    }

    /** TCP-connect к SOCKS-порту: только флагу нет доверия, нужен живой слушатель. */
    private fun socksViable(port: Int): Boolean = try {
        java.net.Socket().use {
            it.connect(java.net.InetSocketAddress(LOOPBACK, port), SOCKS_PROBE_TIMEOUT_MS)
        }
        true
    } catch (e: Exception) {
        false
    }

    private fun isRunningLocked(): Boolean {
        val response = LibXray.invoke(
            """{"apiVersion":3,"method":"getXrayState","payload":{}}"""
        )
        val running = JSONObject(response)
            .optJSONObject("data")?.optBoolean("running", false) ?: false
        Log.i(LOG_TAG, "getXrayState -> $response (parsed running=$running)")
        return running
    }

    fun isRunning(): Boolean = synchronized(lifecycleLock) { isRunningLocked() }

    fun version(): String = JSONObject(
        LibXray.invoke("""{"apiVersion":3,"method":"xrayVersion","payload":{}}""")
    ).optJSONObject("data")?.optString("version", "unknown") ?: "unknown"

    private fun pickFreePort(): Result<Int> {
        val response = invoke(
            "getFreePorts",
            JSONObject().put("count", PORT_CANDIDATES).toString()
        ).getOrElse { return Result.failure(it) }
        val ports = JSONObject(response).getJSONObject("data").getJSONArray("ports")
        for (i in 0 until ports.length()) {
            val candidate = ports.getInt(i)
            if (isBindable(candidate)) {
                return Result.success(candidate)
            }
        }
        return Result.failure(IllegalStateException("нет свободного порта среди кандидатов"))
    }

    private fun isBindable(port: Int): Boolean = try {
        java.net.ServerSocket(port, 1, java.net.InetAddress.getByName(LOOPBACK)).use { true }
    } catch (e: Exception) {
        false
    }

    private fun buildVlessConfig(port: Int): String {
        val vlessOutbound = JSONObject()
            .put("tag", "vless-out")
            .put("protocol", "vless")
            .put(
                "settings",
                JSONObject().put("vnext", JSONArray().put(
                    JSONObject()
                        .put("address", SERVER_ADDRESS)
                        .put("port", SERVER_PORT)
                        .put("users", JSONArray().put(
                            JSONObject()
                                .put("id", USER_UUID)
                                .put("encryption", "none")
                        ))
                ))
            )
            .put(
                "streamSettings",
                JSONObject()
                    .put("network", "ws")
                    .put("security", "tls")
                    .put(
                        "tlsSettings",
                        JSONObject()
                            .put("serverName", SERVER_NAME)
                            .put("allowInsecure", false)
                    )
                    .put(
                        "wsSettings",
                        JSONObject()
                            .put("path", WS_PATH)
                            .put("headers", JSONObject().put("Host", WS_HOST))
                    )
            )

        return JSONObject()
            .put("log", JSONObject().put("loglevel", logLevel))
            // DNS резолвится самим Xray и только в IPv4: имя не отваливается,
            // и приложение не зависает на недостижимых IPv6-адресах.
            .put(
                "dns",
                JSONObject()
                    .put("servers", JSONArray().put("1.1.1.1").put("8.8.8.8"))
                    .put("queryStrategy", "UseIPv4")
            )
            .put(
                "inbounds",
                JSONArray().put(
                    JSONObject()
                        .put("tag", "socks-in")
                        .put("protocol", "socks")
                        .put("listen", LOOPBACK)
                        .put("port", port)
                        // UDP обязателен: на нём работают DNS, QUIC и звонки WhatsApp.
                        .put("settings", JSONObject().put("auth", "noauth").put("udp", true))
                        .put("sniffing", JSONObject().put("enabled", false))
                )
            )
            .put(
                "outbounds",
                JSONArray()
                    .put(vlessOutbound)
                    .put(
                        JSONObject()
                            .put("tag", "dns-out")
                            .put("protocol", "dns")
                            .put(
                                "settings",
                                JSONObject().put(
                                    "rules",
                                    JSONArray().put(
                                        JSONObject()
                                            .put("action", "return")
                                            .put("qType", 28)
                                    )
                                )
                            )
                    )
            )
            // Все DNS-запросы отдаём самому Xray, а не выпускаем наружу как UDP.
            .put(
                "routing",
                JSONObject()
                    .put("domainStrategy", "AsIs")
                    .put(
                        "rules",
                        JSONArray().put(
                            JSONObject()
                                .put("type", "field")
                                .put("port", "53")
                                .put("outboundTag", "dns-out")
                        )
                    )
            )
            .toString()
    }
    private fun invoke(method: String, payload: String): Result<String> {
        val request = """{"apiVersion":3,"method":"$method","payload":$payload}"""
        val response = LibXray.invoke(request)
        val parsed = JSONObject(response)
        return if (parsed.optBoolean("success", false)) {
            Result.success(response)
        } else {
            Result.failure(IllegalStateException("libXray $method: ${parsed.optString("error")}"))
        }
    }

    private fun resetDNS() {
        runCatching { LibXray.resetDNS() }
    }

    private const val LOOPBACK = "127.0.0.1"
    private const val PORT_CANDIDATES = 4
    private const val RUNNING_TIMEOUT_MS = 3_000L
    private const val RUNNING_POLL_MS = 100L
    private const val RESTART_ATTEMPTS = 3
    private const val RESTART_QUIESCE_MS = 400L
    private const val RESTART_STABILIZE_MS = 700L
    private const val SOCKS_PROBE_TIMEOUT_MS = 300

    private const val SERVER_ADDRESS = "ru.netdanet.ru"
    private const val SERVER_PORT = 443
    private const val SERVER_NAME = "ru.netdanet.ru"
    private const val WS_PATH = "/ws"
    private const val WS_HOST = "ru.netdanet.ru"
    private const val USER_UUID = "c012f3d5-e9f3-485a-b406-436fb2485dfa"

===== app\src\test\java\com\kaitag\vpn\NetworkHandoverDeciderFlowTest.kt =====
package com.kaitag.vpn

import com.kaitag.vpn.NetworkHandoverDecider.Decision
import com.kaitag.vpn.NetworkHandoverDecider.Snapshot
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test

/**
 * JVM-тесты end-to-end-контракта дебаунс-коммита: сначала сеть фиксируется
 * (Adopt), при смене сети планируется Pending, а реальный перезапуск отдаётся
 * ровно один раз — в maintenance-раунде по истечении задержки.
 */
class NetworkHandoverDeciderFlowTest {

    private val wifi = { id: Long -> Snapshot(id, "wifi", hasInternet = true, validated = true) }
    private val cellular = { id: Long -> Snapshot(id, "cellular", hasInternet = true, validated = true) }

    private fun unusable(id: Long) = Snapshot(id, "wifi", hasInternet = false, validated = false)

    @Test
    fun `first usable network is adopted and remembered`() {
        val d = NetworkHandoverDecider()
        val r = d.onDefaultNetwork(wifi(1))
        assertTrue(r is Decision.Adopt)
        assertEquals(1L, (r as Decision.Adopt).networkId)
        assertTrue(d.hasCurrent)
        assertFalse(d.pending)
    }

    @Test
    fun `same network after adopt is ignored`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        val r = d.onDefaultNetwork(wifi(1))
        assertEquals(Decision.Ignore, r)
    }

    @Test
    fun `unusable network during startup is ignored`() {
        val d = NetworkHandoverDecider()
        val r = d.onDefaultNetwork(unusable(1))
        assertEquals(Decision.Ignore, r)
        assertFalse(d.hasCurrent)
    }

    @Test
    fun `network change schedules a restart and sets pending`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        val r = d.onDefaultNetwork(cellular(2))
        assertTrue(r is Decision.ScheduleRestart)
        val restart = r as Decision.ScheduleRestart
        assertEquals(2L, restart.networkId)
        assertEquals("wifi", restart.fromType)
        assertEquals("cellular", restart.toType)
        assertTrue(d.pending)
    }

    @Test
    fun `same pending network before debounce is ignored without clearing the timer`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        val r = d.onDefaultNetwork(cellular(2))
        assertEquals(Decision.Ignore, r)
        assertTrue(d.pending)
    }

    @Test
    fun `same pending network at maintenance round commits a single restart`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        val r = d.onDefaultNetwork(cellular(2), maintenance = true)
        assertTrue(r is Decision.ScheduleRestart)
        assertEquals(2L, (r as Decision.ScheduleRestart).networkId)
    }

    @Test
    fun `returning to the previous network cancels the pending change`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        val r = d.onDefaultNetwork(wifi(1))
        assertEquals(Decision.CancelPending(1L, "wifi"), r)
        assertFalse(d.pending)
    }

    @Test
    fun `third network while pending repoints the restart target`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        val r = d.onDefaultNetwork(wifi(3))
        assertTrue(r is Decision.ScheduleRestart)
        assertEquals(3L, (r as Decision.ScheduleRestart).networkId)
        assertTrue(d.pending)
    }

    @Test
    fun `maintenance round with unusable network clears a stuck pending`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        val r = d.onDefaultNetwork(unusable(2), maintenance = true)
        assertEquals(Decision.Ignore, r)
        assertFalse(d.pending)
    }

    @Test
    fun `completed restart is committed and a fresh change schedules again`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        d.onRestartCompleted(2L, "cellular")
        assertEquals(Decision.Ignore, d.onDefaultNetwork(cellular(2)))
        val r = d.onDefaultNetwork(wifi(10))
        assertTrue(r is Decision.ScheduleRestart)
    }

    @Test
    fun `failed restart clears pending so the next suitable callback can retry`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        d.onRestartFailed()
        assertFalse(d.pending)
        assertTrue(d.hasCurrent)
        val retry = d.onDefaultNetwork(cellular(2))
        assertTrue(retry is Decision.ScheduleRestart)
        assertEquals(2L, (retry as Decision.ScheduleRestart).networkId)
    }

    @Test
    fun `failed restart does not reschedule a restart on the original network`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        d.onRestartFailed()
        assertEquals(Decision.Ignore, d.onDefaultNetwork(wifi(1)))
    }

    @Test
    fun `reset makes the next usable network an adoption again`() {
        val d = NetworkHandoverDecider()
        d.onDefaultNetwork(wifi(1))
        d.onDefaultNetwork(cellular(2))
        d.reset()
        assertFalse(d.hasCurrent)
        val r = d.onDefaultNetwork(wifi(1))
        assertTrue(r is Decision.Adopt)
    }
}

import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test
import java.util.concurrent.CopyOnWriteArrayList
import java.util.concurrent.CountDownLatch
import java.util.concurrent.Executors
import java.util.concurrent.TimeUnit

/**
 * Regression coverage for the Mobile <-> Wi-Fi handover decision logic.
 *
 * Covers the rule set behind the "Wi-Fi gets ! after switching with VPN on" bug:
 * one restart per real default-network change, no restarts on capability churn,
 * no restart right after the initial VPN start.
 */
class NetworkHandoverDeciderTest {

    private fun snapshot(
        id: Long,
        type: String = "wifi",
        internet: Boolean = true,
        validated: Boolean = true,
    ) = NetworkHandoverDecider.Snapshot(id, type, internet, validated)

    // ---------------------------------------------------------------- initial start

    @Test
    fun `initial start adopts network without restart`() {
        val decider = NetworkHandoverDecider()
        val decision = decider.onDefaultNetwork(snapshot(1L, "cellular"))
        assertTrue(
            "first observation must be Adopt, got $decision",
            decision is NetworkHandoverDecider.Decision.Adopt,
        )
        assertFalse("no restart may be pending after initial adopt", decider.pending)
    }

    @Test
    fun `initial start ignores unvalidated network`() {
        val decider = NetworkHandoverDecider()
        val decision = decider.onDefaultNetwork(snapshot(1L, validated = false))
        assertEquals(NetworkHandoverDecider.Decision.Ignore, decision)
        assertFalse("adopting nothing must not leave a pending restart", decider.pending)
    }

    @Test
    fun `network without internet capability is ignored`() {
        val decider = NetworkHandoverDecider()
        val decision = decider.onDefaultNetwork(snapshot(1L, internet = false))
        assertEquals(NetworkHandoverDecider.Decision.Ignore, decision)
    }

    @Test
    fun `null snapshot is ignored`() {
        val decider = NetworkHandoverDecider()
        assertEquals(NetworkHandoverDecider.Decision.Ignore, decider.onDefaultNetwork(null))
    }

    // ---------------------------------------------------------------- no spurious restarts

    @Test
    fun `unchanged network produces no restart`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        repeat(5) {
            assertEquals(
                NetworkHandoverDecider.Decision.Ignore,
                decider.onDefaultNetwork(snapshot(1L, "cellular")),
            )
        }
        assertFalse(decider.pending)
    }

    @Test
    fun `capability churn on the same network produces no restart`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "wifi"))
        // onCapabilitiesChanged вызывается многократно, в том числе без validation.
        assertEquals(NetworkHandoverDecider.Decision.Ignore, decider.onDefaultNetwork(snapshot(1L, "wifi", validated = false)))
        assertEquals(NetworkHandoverDecider.Decision.Ignore, decider.onDefaultNetwork(snapshot(1L, "wifi", validated = true)))
        assertEquals(NetworkHandoverDecider.Decision.Ignore, decider.onDefaultNetwork(snapshot(1L, "wifi")))
        assertFalse("capability updates must not schedule a restart", decider.pending)
    }

    // ---------------------------------------------------------------- real handover

    @Test
    fun `real default network change schedules exactly one restart`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))

        val decision = decider.onDefaultNetwork(snapshot(2L, "wifi"))
        assertTrue(
            "new validated default network must schedule a restart, got $decision",
            decision is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        assertEquals(2L, (decision as NetworkHandoverDecider.Decision.ScheduleRestart).networkId)
        assertTrue(decider.pending)
    }

    @Test
    fun `multiple callbacks during transition do not schedule parallel restarts`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))

        val first = decider.onDefaultNetwork(snapshot(2L, "wifi"))
        assertTrue(first is NetworkHandoverDecider.Decision.ScheduleRestart)

        // Handover генерирует ещё несколько событий по той же новой сети
        // (available / capabilities / lost): они не должны сбрасывать задержку
        // и не должны порождать второй перезапуск.
        repeat(4) {
            assertEquals(
                NetworkHandoverDecider.Decision.Ignore,
                decider.onDefaultNetwork(snapshot(2L, "wifi")),
            )
        }
    }

    @Test
    fun `confirmation after debounce performs the restart once`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        decider.onDefaultNetwork(snapshot(2L, "wifi"))

        val confirmed = decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
        assertTrue(
            "debounced confirmation must restart the data path, got $confirmed",
            confirmed is NetworkHandoverDecider.Decision.ScheduleRestart,
        )

        decider.onRestartCompleted(2L, "wifi")
        assertFalse("pending flag must be cleared after the restart", decider.pending)
        // Повторная проверка после перезапуска не должна запускать второй.
        assertEquals(
            NetworkHandoverDecider.Decision.Ignore,
            decider.onDefaultNetwork(snapshot(2L, "wifi")),
        )
    }

    @Test
    fun `network changed again while pending restarts on the newest network`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        decider.onDefaultNetwork(snapshot(2L, "wifi"))

        // Пока шла задержка, default network сменилась ещё раз.
        val decision = decider.onDefaultNetwork(snapshot(3L, "wifi"))
        assertTrue(decision is NetworkHandoverDecider.Decision.ScheduleRestart)
        assertEquals(3L, (decision as NetworkHandoverDecider.Decision.ScheduleRestart).networkId)
    }

    @Test
    fun `returning to the previous network cancels the pending restart`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        decider.onDefaultNetwork(snapshot(2L, "wifi"))

        val decision = decider.onDefaultNetwork(snapshot(1L, "cellular"))
        assertTrue(
            "coming back to the current network must cancel the restart, got $decision",
            decision is NetworkHandoverDecider.Decision.CancelPending,

        assertFalse(decider.pending)
    }

    @Test
    fun `failed restart clears pending so a later change can retry`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        decider.onDefaultNetwork(snapshot(2L, "wifi"))
        decider.onRestartFailed()
        assertFalse(decider.pending)

        val retry = decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
        assertTrue(
            "after a failed restart the same network may be retried once, got $retry",
            retry is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
    }

        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        decider.reset()

        val decision = decider.onDefaultNetwork(snapshot(2L, "wifi"))
        assertTrue(
            "after reset the first network must be adopted, not restarted",
            decision is NetworkHandoverDecider.Decision.Adopt,
        )
    }

        )
        assertTrue("restart must be pending before the confirmation", decider.pending)

        // Окно подтверждения вскрылось, а сеть всё ещё не validated.
        assertEquals(
            NetworkHandoverDecider.Decision.Ignore,
            decider.onDefaultNetwork(snapshot(2L, "wifi", validated = false), maintenance = true),
        )
        assertFalse("a failed confirmation must not leave a hung pending state", decider.pending)

        // Как только B стала usable — handover планируется заново и выполняется ровно один раз.
        val rescheduled = decider.onDefaultNetwork(snapshot(2L, "wifi", validated = true))
        assertTrue(
            "the network that became usable must schedule the restart again, got $rescheduled",
            rescheduled is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        assertEquals(2L, (rescheduled as NetworkHandoverDecider.Decision.ScheduleRestart).networkId)
        val confirmed = decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
        assertTrue(confirmed is NetworkHandoverDecider.Decision.ScheduleRestart)
        decider.onRestartCompleted(2L, "wifi")
        assertFalse(decider.pending)
    }

    @Test
    fun `confirmation with no network at all does not hang the handover`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        assertTrue(
            decider.onDefaultNetwork(snapshot(2L, "wifi"))
                is NetworkHandoverDecider.Decision.ScheduleRestart,
        )

        // В момент подтверждения usable-сети нет вообще (B пропала, A ещё не готова).
        assertEquals(
            NetworkHandoverDecider.Decision.Ignore,
            decider.onDefaultNetwork(null, maintenance = true),
        )
        assertFalse("no usable network at confirmation must clear the pending restart", decider.pending)

        // Появление usable B снова приводит к полноценному handover, без зависшего pending.
        val rescheduled = decider.onDefaultNetwork(snapshot(2L, "wifi"))
        assertTrue(
            "a usable network appearing later must retry the handover, got $rescheduled",
            rescheduled is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        assertTrue(
            decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
                is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        decider.onRestartCompleted(2L, "wifi")
        assertFalse(decider.pending)
    }

    // --------------------------------------------------- regression: быстрые флапы A -> B -> A -> B

    @Test
    fun `rapid A to B to A to B flapping never runs parallel restarts`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))

        // Первый уход на B вооружает ровно один pending restart.
        assertTrue(
            decider.onDefaultNetwork(snapshot(2L, "wifi"))
                is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        assertTrue("one pending restart after the flap to B", decider.pending)

        // Шквал callbacks по той же B до debounce не сбрасывает задержку и не
        // создаёт второго restart.
        repeat(4) {
            assertEquals(
                NetworkHandoverDecider.Decision.Ignore,
                decider.onDefaultNetwork(snapshot(2L, "wifi")),
            )
        }

        // Быстрый возврат на A отменяет единственный pending restart.
        assertTrue(
            decider.onDefaultNetwork(snapshot(1L, "cellular"))
                is NetworkHandoverDecider.Decision.CancelPending,
        )
        assertFalse("returning to A must clear the pending restart", decider.pending)

        // Снова уход на B (второй цикл флапа) — снова ровно один pending, ни
        // одного «застрявшего» restart от первого цикла.
        assertTrue(
            decider.onDefaultNetwork(snapshot(2L, "wifi"))
                is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        assertTrue(decider.pending)

        // Пока restart стоит по B, повторные callbacks B — только Ignore.
        repeat(3) {
            assertEquals(
                NetworkHandoverDecider.Decision.Ignore,
                decider.onDefaultNetwork(snapshot(2L, "wifi")),
            )
        }

        // Флап прекратился, сеть осела на B: подтверждаем ровно один перезапуск.
        assertTrue(
            decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
                is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        decider.onRestartCompleted(2L, "wifi")
        assertFalse("a settled flap leaves nothing pending", decider.pending)

        // После перезапуска повторные callbacks B (capabilities/lost) не дают
        // второго restart'а.
        repeat(4) {
            assertEquals(
                NetworkHandoverDecider.Decision.Ignore,
                decider.onDefaultNetwork(snapshot(2L, "wifi")),
            )
        }
        assertFalse("flapping must not leave a second restart armed", decider.pending)
    }

    // ---------------------------------------------- regression: неудачный рестарт -> повторная попытка

    @Test
    fun `failed handover restart is retried by the next suitable callback`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))
        decider.onDefaultNetwork(snapshot(2L, "wifi"))
        assertTrue(
            decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
                is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        // Перезапуск завершился ошибкой.
        decider.onRestartFailed()
        assertFalse("a failed restart must clear the pending state", decider.pending)

        // Next suitable callback (post-restart re-evaluation) планирует retry заново.
        val retry = decider.onDefaultNetwork(snapshot(2L, "wifi"))
        assertTrue(
            "the next suitable callback must allow retrying the failed restart, got $retry",
            retry is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        assertEquals(2L, (retry as NetworkHandoverDecider.Decision.ScheduleRestart).networkId)

        val confirmed = decider.onDefaultNetwork(snapshot(2L, "wifi"), maintenance = true)
        assertTrue(confirmed is NetworkHandoverDecider.Decision.ScheduleRestart)
        decider.onRestartCompleted(2L, "wifi")
        assertFalse(decider.pending)
    }

    // ------------------------------------------- regression: потокобезопасность

    @Test
    fun `decider stays consistent under concurrent callbacks and restarts`() {
        val decider = NetworkHandoverDecider()
        decider.onDefaultNetwork(snapshot(1L, "cellular"))

        val pool = Executors.newFixedThreadPool(8)
        val startGate = CountDownLatch(1)
        val errors = CopyOnWriteArrayList<Throwable>()
        try {
            repeat(8) { worker ->
                pool.submit {
                    startGate.await()
                    runCatching {
                        repeat(1000) { i ->
                            when (i % 4) {
                                0 -> decider.onDefaultNetwork(snapshot(2L, "wifi"))
                                1 -> decider.onDefaultNetwork(snapshot(1L, "cellular"))
                                2 -> decider.onRestartCompleted(2L, "wifi")
                                else -> decider.onRestartFailed()
                            }
                        }
                    }.onFailure { errors.add(it) }
                }
            }
            startGate.countDown()
            pool.shutdown()
            assertTrue(
                "workers did not finish in time",
                pool.awaitTermination(30, TimeUnit.SECONDS),
            )
        } finally {
            pool.shutdownNow()
        }
        assertTrue("no exceptions expected under contention, got $errors", errors.isEmpty())

        // После завершения набега логика всё ещё работает штатно.
        val decision = decider.onDefaultNetwork(snapshot(3L, "wifi"))
        assertTrue(
            "decider must still schedule a handover, got $decision",
            decision is NetworkHandoverDecider.Decision.ScheduleRestart,
        )
        decider.onRestartCompleted(3L, "wifi")
        assertFalse(decider.pending)
    }
